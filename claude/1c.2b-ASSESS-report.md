    75	/**
    76	 * Walk an arbitrary nested object, collecting leaf nodes whose values are
    77	 * finite numbers. Rules:
    78	 *   - null/undefined → skip
    79	 *   - arrays → skip entirely (produce nothing)
    80	 *   - key named exactly "year" → skip
    81	 *   - finite number → emit a figure
    82	 *   - object (including empty) → recurse; empty objects yield nothing naturally
    83	 */
    84	function walkForFigures(
    85	  obj: unknown,
    86	  path: string,
    87	  figures: EvidenceFigure[],
    88	): void {
    89	  if (obj === null || obj === undefined) return;
    90	  if (Array.isArray(obj)) return;
    91	  if (typeof obj !== 'object') return;
    92	
    93	  for (const key of Object.keys(obj as Record<string, unknown>)) {
    94	    if (key === 'year') continue;
    95	
    96	    const childPath = `${path}.${key}`;
    97	    const childVal = (obj as Record<string, unknown>)[key];
