# Prompt 2: Value Creation Modeller — Layout Shell + Assumptions Panel

## Context

This is Prompt 2 of the Value Creation Modeller build. Prompt 1 created the database schema (4 tables), calculation engine (`shared/calculators/value-creation.calc.ts`), backend service, and API routes at `/api/v2/value-creation/*`. All of that is working.

Now we're building the front-end shell: a new "Value Creation" tab on the deal page, the two-column layout, and the full assumptions input panel. The right-hand output area will be a placeholder for now — Prompts 3-9 will fill it with charts and tables.

**Existing codebase patterns to follow:**
- The deal page is `client/src/pages/target-company-detail.tsx` — it uses `<Tabs>` with `<TabsList>` and `<TabsTrigger>` from `@/components/ui/tabs`
- The existing Deal Modeller component is at `client/src/components/target-company/DealModeller.tsx` — it shows the pattern for fetching deal model data, selecting models, and rendering sections
- Data fetching uses `useQuery` from `@tanstack/react-query` with `queryKey` arrays
- Mutations use `useMutation` with `apiRequest('PUT', url, body)` from `@/lib/queryClient`
- The debounced auto-save pattern is in `ValuationSection.tsx` — local state + debounceRef + pendingRef + 600ms timeout
- Format helpers: `formatCurrency` and `formatPercent` from `@/lib/format-utils`
- The `toNum()` helper is in `@shared/calculators/deal-model.calc`
- UI components available: Card, Input, Select, Slider, Badge, Separator, Tabs, Collapsible, Tooltip, Switch, Skeleton, ScrollArea, Button, Sheet (for mobile drawer)
- Lucide icons are used throughout

**API endpoints available (from Prompt 1):**
- `GET /api/v2/value-creation/deal-model/:dealModelId` → returns `{ scenarios, assumptions, synergies, exitAssumptions }`
- `PUT /api/v2/value-creation/assumptions/:id` → update single assumption
- `PUT /api/v2/value-creation/assumptions/bulk` → bulk update `{ updates: [{ id, data }] }`
- `PUT /api/v2/value-creation/scenarios/:id` → update scenario multipliers
- `PUT /api/v2/value-creation/exit/:dealModelId` → update exit assumptions

**Calculation engine imports available:**


### Step 2: Create the Main Component

Create a new file at `client/src/components/target-company/ValueCreationModeller.tsx`:

This is the main orchestrator component. It handles:
- Deal model selection (reuses existing deal model selector)
- Fetching value creation data from the API
- Constructing base year data from financials
- Running the calculation engine
- Two-column layout: assumptions panel (left) + output sections (right)

```typescript
import { useState, useMemo, useEffect, useCallback, useRef } from 'react';
import { useQuery, useMutation } from '@tanstack/react-query';
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
import { Button } from '@/components/ui/button';
import { Badge } from '@/components/ui/badge';
import { Skeleton } from '@/components/ui/skeleton';
import { Separator } from '@/components/ui/separator';
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
import { Tabs, TabsList, TabsTrigger } from '@/components/ui/tabs';
import { Sheet, SheetContent, SheetTrigger } from '@/components/ui/sheet';
import { ScrollArea } from '@/components/ui/scroll-area';
import { useToast } from '@/hooks/use-toast';
import { apiRequest, queryClient } from '@/lib/queryClient';
import { formatCurrency, formatPercent } from '@/lib/format-utils';
import { toNum } from '@shared/calculators/deal-model.calc';
import {
  calculateProjections,
  calculateDebtSchedules,
  calculateWaterfall,
  calculateReturns,
  type BaseYearData,
  type YearAssumptions,
  type SynergyInput,
  type FundingLineForProjection,
  type ExitAssumptions as ExitAssumptionsType,
  type ScenarioMultipliers,
  type ProjectedYear,
} from '@shared/calculators/value-creation.calc';
import type {
  DealModel,
  DealModelFundingLine,
  ValueCreationScenario,
  ValueCreationAssumption,
  ValueCreationSynergy,
  ValueCreationExitAssumption,
} from '@shared/schema';
import AssumptionsPanel from '@/components/value-creation/AssumptionsPanel';
import {
  LineChart,
  Calculator,
  Loader2,
  PanelLeftClose,
  PanelLeftOpen,
  TrendingUp,
  Shield,
  DollarSign,
  BarChart3,
} from 'lucide-react';

interface ValueCreationModellerProps {
  companyId: number;
}

interface DealModelWithDetails extends DealModel {
  adjustments: any[];
  fundingLines: DealModelFundingLine[];
}

interface ValueCreationData {
  scenarios: ValueCreationScenario[];
  assumptions: ValueCreationAssumption[];
  synergies: ValueCreationSynergy[];
  exitAssumptions: ValueCreationExitAssumption | null;
}

export default function ValueCreationModeller({ companyId }: ValueCreationModellerProps) {
  const { toast } = useToast();
  const [selectedModelId, setSelectedModelId] = useState<number | null>(null);
  const [activeScenarioType, setActiveScenarioType] = useState<string>('base');
  const [panelCollapsed, setPanelCollapsed] = useState(false);

  // ─── Fetch deal models (same pattern as DealModeller.tsx) ───
  const { data: models = [], isLoading: modelsLoading } = useQuery<DealModel[]>({
    queryKey: ['/api/v2/deal-models/company', companyId],
    select: (response: any) => response?.data || response || [],
    enabled: !!companyId,
  });

  useEffect(() => {
    if (models.length > 0 && selectedModelId === null) {
      setSelectedModelId(models[0].id);
    }
  }, [models, selectedModelId]);

  // Fetch deal model details (for funding lines + base financial data)
  const { data: modelDetails } = useQuery<DealModelWithDetails>({
    queryKey: ['/api/v2/deal-models', selectedModelId],
    select: (response: any) => response?.data || response,
    enabled: !!selectedModelId,
  });

  // Fetch financial data (for base year P&L)
  const { data: financialsRaw } = useQuery<any>({
    queryKey: ['/api/v2/financial-data/company', companyId, 'financials'],
    select: (response: any) => response?.data || response,
    enabled: !!companyId,
  });

  // ─── Fetch value creation data ───
  const { data: vcData, isLoading: vcLoading } = useQuery<ValueCreationData>({
    queryKey: ['/api/v2/value-creation/deal-model', selectedModelId],
    select: (response: any) => response?.data || response,
    enabled: !!selectedModelId,
  });

  // ─── Derive base year data from most recent financial period ───
  const baseYearData = useMemo<BaseYearData | null>(() => {
    if (!financialsRaw?.periods || !financialsRaw?.incomeStatements) return null;

    // Get the most recent statutory period
    const statutory = financialsRaw.periods
      .filter((p: any) => (p.periodType === 'statutory' || p.periodType === 'historic') && p.fiscalYear)
      .sort((a: any, b: any) => new Date(b.periodEnd).getTime() - new Date(a.periodEnd).getTime());

    if (statutory.length === 0) return null;

    const latestPeriod = statutory[0];
    const income = financialsRaw.incomeStatements.find(
      (i: any) => i.financialPeriodId === latestPeriod.id
    );

    if (!income) return null;

    const revenue = toNum(income.revenue);
    const costOfSales = toNum(income.costOfSales);
    const grossProfit = toNum(income.grossProfit) || (revenue - costOfSales);
    const adminExpenses = toNum(income.administrativeExpenses);
    const staffCosts = toNum(income.staffCosts);
    const depreciation = toNum(income.depreciationAmortisation);
    const operatingProfit = toNum(income.operatingProfit);
    const interestPayable = toNum(income.interestPayable);
    const corporationTax = toNum(income.corporationTax);
    const netProfit = toNum(income.netProfit);
    const adjustedEbitda = operatingProfit + depreciation;

    return {
      revenue,
      costOfSales,
      grossProfit,
      administrativeExpenses: adminExpenses,
      staffCosts,
      depreciationAmortisation: depreciation,
      operatingProfit,
      interestPayable,
      corporationTax,
      netProfit,
      adjustedEbitda,
    };
  }, [financialsRaw]);

  // ─── Get the base year label ───
  const baseYearLabel = useMemo<string>(() => {
    if (!financialsRaw?.periods) return 'Base';
    const statutory = financialsRaw.periods
      .filter((p: any) => (p.periodType === 'statutory' || p.periodType === 'historic') && p.fiscalYear)
      .sort((a: any, b: any) => new Date(b.periodEnd).getTime() - new Date(a.periodEnd).getTime());
    return statutory[0]?.fiscalYear ? `FY${statutory[0].fiscalYear}` : 'Base';
  }, [financialsRaw]);

  // ─── Get active scenario and its assumptions ───
  const activeScenario = useMemo(() => {
    if (!vcData?.scenarios) return null;
    return vcData.scenarios.find(s => s.scenarioType === activeScenarioType) || vcData.scenarios[0];
  }, [vcData, activeScenarioType]);

  const activeAssumptions = useMemo<YearAssumptions[]>(() => {
    if (!vcData?.assumptions || !activeScenario) return [];
    return vcData.assumptions
      .filter(a => a.scenarioId === activeScenario.id)
      .sort((a, b) => a.year - b.year)
      .map(a => ({
        year: a.year,
        revenueGrowthPercent: toNum(a.revenueGrowthPercent),
        costOfSalesGrowthPercent: toNum(a.costOfSalesGrowthPercent),
        adminCostGrowthPercent: toNum(a.adminCostGrowthPercent),
        staffCostGrowthPercent: toNum(a.staffCostGrowthPercent),
        depreciationGrowthPercent: toNum(a.depreciationGrowthPercent),
        capexPerYear: toNum(a.capexPerYear),
        workingCapitalPercent: toNum(a.workingCapitalPercent),
        taxRatePercent: toNum(a.taxRatePercent),
      }));
  }, [vcData, activeScenario]);

  // ─── Build synergy inputs ───
  const synergyInputs = useMemo<SynergyInput[]>(() => {
    if (!vcData?.synergies || !activeScenario) return [];
    return vcData.synergies
      .filter(s => s.scenarioId === activeScenario.id)
      .map(s => ({
        synergyType: s.synergyType as 'revenue' | 'cost',
        name: s.name,
        targetValue: toNum(s.targetValue),
        realisationByYear: [
          toNum(s.realisationY1) / 100,
          toNum(s.realisationY2) / 100,
          toNum(s.realisationY3) / 100,
          toNum(s.realisationY4) / 100,
          toNum(s.realisationY5) / 100,
        ],
        implementationCost: toNum(s.implementationCost),
      }));
  }, [vcData, activeScenario]);

  // ─── Build funding line inputs ───
  const fundingLineInputs = useMemo<FundingLineForProjection[]>(() => {
    if (!modelDetails?.fundingLines) return [];
    return modelDetails.fundingLines.map(f => ({
      amountUsed: toNum(f.amountUsed),
      interestRate: toNum(f.interestRate),
      termMonths: f.termMonths ?? 60,
      isDebt: f.isDebt ?? false,
      isActive: f.isActive ?? false,
    }));
  }, [modelDetails]);

  // ─── Build exit assumptions ───
  const exitAssumptions = useMemo<ExitAssumptionsType>(() => {
    if (!vcData?.exitAssumptions) {
      return { exitMultiple: 4.0, holdPeriodYears: 5, exitCostPercent: 3.0 };
    }
    return {
      exitMultiple: toNum(vcData.exitAssumptions.exitMultiple),
      holdPeriodYears: vcData.exitAssumptions.holdPeriodYears ?? 5,
      exitCostPercent: toNum(vcData.exitAssumptions.exitCostPercent),
    };
  }, [vcData]);

  // ─── Run projections ───
  const projections = useMemo<ProjectedYear[]>(() => {
    if (!baseYearData || activeAssumptions.length === 0) return [];
    return calculateProjections(baseYearData, activeAssumptions, synergyInputs, fundingLineInputs);
  }, [baseYearData, activeAssumptions, synergyInputs, fundingLineInputs]);

  // ─── Computed KPIs for the global strip ───
  const kpis = useMemo(() => {
    if (projections.length === 0) return null;
    const y5 = projections[projections.length - 1];
    const totalCash = projections.reduce((sum, p) => sum + p.freeCashFlow, 0);
    const avgDscr = projections.reduce((sum, p) => sum + p.dscr, 0) / projections.length;
    const worstDscr = Math.min(...projections.map(p => p.dscr));
    return {
      y5Revenue: y5.revenue,
      y5Ebitda: y5.ebitda,
      totalCashGenerated: totalCash,
      avgDscr,
      worstDscr,
    };
  }, [projections]);

  // ─── Callback for when assumptions panel saves ───
  const handleDataChange = useCallback(() => {
    if (selectedModelId) {
      queryClient.invalidateQueries({ queryKey: ['/api/v2/value-creation/deal-model', selectedModelId] });
    }
  }, [selectedModelId]);

  // ─── Deal context for display ───
  const entryMultiple = toNum(modelDetails?.ebitdaMultiple);
  const entryEV = toNum(modelDetails?.enterpriseValue);
  const totalDebt = fundingLineInputs
    .filter(f => f.isDebt && f.isActive)
    .reduce((sum, f) => sum + f.amountUsed, 0);

  // ─── No deal models state ───
  if (modelsLoading) {
    return (
      <div className="space-y-4">
        <Skeleton className="h-12 w-full" />
        <Skeleton className="h-96 w-full" />
      </div>
    );
  }

  if (models.length === 0) {
    return (
      <Card>
        <CardContent className="py-10 flex flex-col items-center text-center gap-4">
          <div className="rounded-full bg-primary/10 p-4">
            <LineChart className="h-8 w-8 text-primary" />
          </div>
          <div>
            <h3 className="text-lg font-semibold">Value Creation Modeller</h3>
            <p className="text-sm text-muted-foreground mt-1 max-w-md">
              Create a deal model first in the Deal Modeller tab. The value creation modeller projects
              what this business looks like over 5 years under your ownership.
            </p>
          </div>
          <Badge variant="secondary">Requires a Deal Model</Badge>
        </CardContent>
      </Card>
    );
  }

  const selectedModel = models.find(m => m.id === selectedModelId);

  // ─── Assumptions panel content (shared between desktop sidebar and mobile sheet) ───
  const assumptionsPanelContent = activeScenario && vcData ? (
    <AssumptionsPanel
      scenario={activeScenario}
      assumptions={vcData.assumptions.filter(a => a.scenarioId === activeScenario.id)}
      synergies={vcData.synergies.filter(s => s.scenarioId === activeScenario.id)}
      exitAssumptions={vcData.exitAssumptions}
      dealModelId={selectedModelId!}
      fundingLines={modelDetails?.fundingLines || []}
      baseYearData={baseYearData}
      onDataChange={handleDataChange}
    />
  ) : null;

  return (
    <div className="space-y-4">
      {/* ─── Global Controls Strip ─── */}
      <div className="flex flex-wrap items-center gap-3">
        {/* Model selector */}
        <Select
          value={selectedModelId ? String(selectedModelId) : undefined}
          onValueChange={(val) => setSelectedModelId(Number(val))}
        >
          <SelectTrigger className="w-[220px]">
            <SelectValue placeholder="Select deal model" />
          </SelectTrigger>
          <SelectContent>
            {models.map(m => (
              <SelectItem key={m.id} value={String(m.id)}>
                {m.name} (v{m.version})
              </SelectItem>
            ))}
          </SelectContent>
        </Select>

        {/* Scenario switcher */}
        <Tabs value={activeScenarioType} onValueChange={setActiveScenarioType}>
          <TabsList className="h-9">
            <TabsTrigger value="bear" className="text-xs px-3 data-[state=active]:bg-red-100 data-[state=active]:text-red-700 dark:data-[state=active]:bg-red-900/30 dark:data-[state=active]:text-red-300">
              🐻 Bear
            </TabsTrigger>
            <TabsTrigger value="base" className="text-xs px-3 data-[state=active]:bg-blue-100 data-[state=active]:text-blue-700 dark:data-[state=active]:bg-blue-900/30 dark:data-[state=active]:text-blue-300">
              📊 Base
            </TabsTrigger>
            <TabsTrigger value="bull" className="text-xs px-3 data-[state=active]:bg-green-100 data-[state=active]:text-green-700 dark:data-[state=active]:bg-green-900/30 dark:data-[state=active]:text-green-300">
              🐂 Bull
            </TabsTrigger>
          </TabsList>
        </Tabs>

        {/* Deal context badges */}
        <div className="flex items-center gap-2 ml-auto">
          {entryMultiple > 0 && (
            <Badge variant="outline" className="text-xs">
              Entry: {entryMultiple.toFixed(1)}x
            </Badge>
          )}
          {totalDebt > 0 && (
            <Badge variant="outline" className="text-xs">
              Debt: {formatCurrency(totalDebt)}
            </Badge>
          )}

          {/* Panel toggle (desktop) */}
          <Button
            variant="ghost"
            size="sm"
            onClick={() => setPanelCollapsed(!panelCollapsed)}
            className="hidden lg:flex"
          >
            {panelCollapsed ? <PanelLeftOpen className="h-4 w-4" /> : <PanelLeftClose className="h-4 w-4" />}
          </Button>

          {/* Panel toggle (mobile — opens sheet) */}
          <Sheet>
            <SheetTrigger asChild>
              <Button variant="outline" size="sm" className="lg:hidden">
                <PanelLeftOpen className="h-4 w-4 mr-2" />
                Assumptions
              </Button>
            </SheetTrigger>
            <SheetContent side="left" className="w-[340px] p-0">
              <ScrollArea className="h-full">
                <div className="p-4">
                  {assumptionsPanelContent}
                </div>
              </ScrollArea>
            </SheetContent>
          </Sheet>
        </div>
      </div>

      {/* ─── KPI Strip ─── */}
      {kpis && (
        <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
          <Card className="border-l-4 border-l-blue-500">
            <CardContent className="py-3 px-4">
              <p className="text-xs text-muted-foreground flex items-center gap-1">
                <TrendingUp className="h-3 w-3" />
                Y5 Revenue
              </p>
              <p className="text-lg font-bold mt-0.5">{formatCurrency(kpis.y5Revenue)}</p>
            </CardContent>
          </Card>
          <Card className="border-l-4 border-l-green-500">
            <CardContent className="py-3 px-4">
              <p className="text-xs text-muted-foreground flex items-center gap-1">
                <BarChart3 className="h-3 w-3" />
                Y5 EBITDA
              </p>
              <p className="text-lg font-bold mt-0.5">{formatCurrency(kpis.y5Ebitda)}</p>
            </CardContent>
          </Card>
          <Card className="border-l-4 border-l-emerald-500">
            <CardContent className="py-3 px-4">
              <p className="text-xs text-muted-foreground flex items-center gap-1">
                <DollarSign className="h-3 w-3" />
                Total Cash Generated
              </p>
              <p className="text-lg font-bold mt-0.5">{formatCurrency(kpis.totalCashGenerated)}</p>
            </CardContent>
          </Card>
          <Card className={`border-l-4 ${
            kpis.worstDscr >= 1.8 ? 'border-l-green-500' :
            kpis.worstDscr >= 1.4 ? 'border-l-blue-500' :
            kpis.worstDscr >= 1.0 ? 'border-l-amber-500' :
            'border-l-red-500'
          }`}>
            <CardContent className="py-3 px-4">
              <p className="text-xs text-muted-foreground flex items-center gap-1">
                <Shield className="h-3 w-3" />
                Worst DSCR
              </p>
              <p className={`text-lg font-bold mt-0.5 ${
                kpis.worstDscr >= 1.8 ? 'text-green-700 dark:text-green-400' :
                kpis.worstDscr >= 1.4 ? 'text-blue-700 dark:text-blue-400' :
                kpis.worstDscr >= 1.0 ? 'text-amber-700 dark:text-amber-400' :
                'text-red-700 dark:text-red-400'
              }`}>
                {kpis.worstDscr >= 100 ? '∞' : kpis.worstDscr.toFixed(2)}x
              </p>
            </CardContent>
          </Card>
        </div>
      )}

      {/* ─── Two-Column Layout ─── */}
      {vcLoading ? (
        <div className="space-y-4">
          <Skeleton className="h-96 w-full" />
        </div>
      ) : (
        <div className="flex gap-4">
          {/* Left: Assumptions Panel (desktop) */}
          {!panelCollapsed && (
            <div className="hidden lg:block w-[320px] shrink-0">
              <div className="sticky top-20">
                <ScrollArea className="h-[calc(100vh-200px)]">
                  {assumptionsPanelContent}
                </ScrollArea>
              </div>
            </div>
          )}

          {/* Right: Output Area */}
          <div className="flex-1 min-w-0 space-y-6">
            {!baseYearData ? (
              <Card>
                <CardContent className="py-10 text-center">
                  <Calculator className="h-8 w-8 mx-auto mb-3 text-muted-foreground" />
                  <h3 className="font-semibold mb-1">No Financial Data</h3>
                  <p className="text-sm text-muted-foreground max-w-md mx-auto">
                    Upload company accounts and run expert analysis to populate the base year data.
                    The value creation modeller needs at least one year of P&L data.
                  </p>
                </CardContent>
              </Card>
            ) : projections.length === 0 ? (
              <Card>
                <CardContent className="py-10 text-center">
                  <Loader2 className="h-8 w-8 mx-auto mb-3 text-muted-foreground animate-spin" />
                  <p className="text-sm text-muted-foreground">Calculating projections...</p>
                </CardContent>
              </Card>
            ) : (
              <>
                {/* Placeholder sections — these will be replaced by Prompts 3-9 */}
                <Card>
                  <CardHeader>
                    <CardTitle className="text-base flex items-center gap-2">
                      <span className="flex items-center justify-center h-6 w-6 rounded-full bg-blue-100 dark:bg-blue-900/30 text-xs font-bold text-blue-700 dark:text-blue-300">1</span>
                      P&L Projections
                    </CardTitle>
                  </CardHeader>
                  <CardContent>
                    <div className="bg-muted/30 rounded-lg p-6 text-center text-sm text-muted-foreground">
                      <p className="font-medium mb-1">Section ready for Prompt 3</p>
                      <p>{baseYearLabel} → Year 5 projections calculated ({projections.length} years)</p>
                      <p className="mt-2 font-mono text-xs">
                        Y1 Revenue: {formatCurrency(projections[0]?.revenue)} → Y5: {formatCurrency(projections[4]?.revenue)}
                      </p>
                      <p className="font-mono text-xs">
                        Y1 EBITDA: {formatCurrency(projections[0]?.ebitda)} → Y5: {formatCurrency(projections[4]?.ebitda)}
                      </p>
                    </div>
                  </CardContent>
                </Card>

                <Card>
                  <CardHeader>
                    <CardTitle className="text-base flex items-center gap-2">
                      <span className="flex items-center justify-center h-6 w-6 rounded-full bg-purple-100 dark:bg-purple-900/30 text-xs font-bold text-purple-700 dark:text-purple-300">4</span>
                      Cash Flow Waterfall
                    </CardTitle>
                  </CardHeader>
                  <CardContent>
                    <div className="bg-muted/30 rounded-lg p-6 text-center text-sm text-muted-foreground">
                      <p className="font-medium">Section ready for Prompt 4</p>
                    </div>
                  </CardContent>
                </Card>

                <Card>
                  <CardHeader>
                    <CardTitle className="text-base flex items-center gap-2">
                      <span className="flex items-center justify-center h-6 w-6 rounded-full bg-emerald-100 dark:bg-emerald-900/30 text-xs font-bold text-emerald-700 dark:text-emerald-300">5</span>
                      Returns Analysis
                    </CardTitle>
                  </CardHeader>
                  <CardContent>
```

```typescript
// ─── NumericStepper: [-] [input] [unit] [+] ───
function NumericStepper({
  value,
  onChange,
  step = 0.5,
  min = -50,
  max = 100,
  unit = '%',
  disabled = false,
}: {
  value: number;
  onChange: (value: number) => void;
  step?: number;
  min?: number;
  max?: number;
  unit?: string;
  disabled?: boolean;
}) {
  const clamp = (v: number) => Math.min(max, Math.max(min, Math.round(v * 100) / 100));

  return (
    <div className="flex items-center gap-0.5">
      <Button
        variant="outline"
        size="icon"
        className="h-7 w-7 shrink-0"
        onClick={() => onChange(clamp(value - step))}
        disabled={disabled || value <= min}
      >
        <span className="text-sm">−</span>
      </Button>
      <div className="relative w-16">
        <Input
          type="number"
          value={value}
          onChange={e => {
            const v = parseFloat(e.target.value);
            if (!isNaN(v)) onChange(clamp(v));
          }}
          disabled={disabled}
          className="h-7 text-xs text-center pr-5 [appearance:textfield] [&::-webkit-outer-spin-button]:appearance-none [&::-webkit-inner-spin-button]:appearance-none"
          step={step}
        />
        <span className="absolute right-1.5 top-1/2 -translate-y-1/2 text-[10px] text-muted-foreground pointer-events-none">
          {unit}
        </span>
      </div>
      <Button
        variant="outline"
        size="icon"
        className="h-7 w-7 shrink-0"
        onClick={() => onChange(clamp(value + step))}
        disabled={disabled || value >= max}
      >
        <span className="text-sm">+</span>
      </Button>
    </div>
  );
}

export default function AssumptionsPanel({
  scenario,
  assumptions,
  synergies,
  exitAssumptions,
  dealModelId,
  fundingLines,
  baseYearData,
  onDataChange,
}: AssumptionsPanelProps) {
  const { toast } = useToast();
  const [perYearMode, setPerYearMode] = useState(false);
  const [growthOpen, setGrowthOpen] = useState(true);
  const [debtOpen, setDebtOpen] = useState(false);
  const [exitOpen, setExitOpen] = useState(true);
  const [saveStatus, setSaveStatus] = useState<'idle' | 'saving' | 'saved'>('idle');
  
  const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const pendingAssumptionUpdates = useRef<Map<number, Record<string, any>>>(new Map());
  const pendingExitUpdates = useRef<Record<string, any> | null>(null);

  const sorted = useMemo(() =>
    [...assumptions].sort((a, b) => a.year - b.year),
    [assumptions]
  );

  // Local state for instant UI updates
  const [localAssumptions, setLocalAssumptions] = useState<Record<number, Record<string, number>>>({});
  const [localExit, setLocalExit] = useState<{
    exitMultiple: number;
    holdPeriodYears: number;
    exitCostPercent: number;
  } | null>(null);

  // Initialise local state from props
  useEffect(() => {
    const map: Record<number, Record<string, number>> = {};
    for (const a of sorted) {
      map[a.id] = {
        revenueGrowthPercent: toNum(a.revenueGrowthPercent),
        costOfSalesGrowthPercent: toNum(a.costOfSalesGrowthPercent),
        adminCostGrowthPercent: toNum(a.adminCostGrowthPercent),
        staffCostGrowthPercent: toNum(a.staffCostGrowthPercent),
        depreciationGrowthPercent: toNum(a.depreciationGrowthPercent),
        capexPerYear: toNum(a.capexPerYear),
        workingCapitalPercent: toNum(a.workingCapitalPercent),
        taxRatePercent: toNum(a.taxRatePercent),
      };
    }
    setLocalAssumptions(map);
  }, [sorted]);

  useEffect(() => {
    if (exitAssumptions) {
      setLocalExit({
        exitMultiple: toNum(exitAssumptions.exitMultiple),
        holdPeriodYears: exitAssumptions.holdPeriodYears ?? 5,
        exitCostPercent: toNum(exitAssumptions.exitCostPercent),
      });
    }
  }, [exitAssumptions]);

  // ─── Mutations ───
  const bulkUpdateMutation = useMutation({
    mutationFn: async (updates: Array<{ id: number; data: Record<string, any> }>) => {
      await apiRequest('PUT', '/api/v2/value-creation/assumptions/bulk', { updates });
    },
    onSuccess: () => {
      setSaveStatus('saved');
      setTimeout(() => setSaveStatus('idle'), 1500);
      onDataChange();
    },
    onError: (error: Error) => {
      setSaveStatus('idle');
      toast({ title: 'Save failed', description: error.message, variant: 'destructive' });
    },
  });

  const exitUpdateMutation = useMutation({
    mutationFn: async (data: Record<string, any>) => {
      await apiRequest('PUT', `/api/v2/value-creation/exit/${dealModelId}`, data);
    },
    onSuccess: () => {
      setSaveStatus('saved');
      setTimeout(() => setSaveStatus('idle'), 1500);
      onDataChange();
    },
    onError: (error: Error) => {
      setSaveStatus('idle');
      toast({ title: 'Save failed', description: error.message, variant: 'destructive' });
    },
  });

  // ─── Debounced save ───
  const flushSave = useCallback(() => {
    if (debounceRef.current) clearTimeout(debounceRef.current);

    const assumptionUpdates = Array.from(pendingAssumptionUpdates.current.entries()).map(
      ([id, data]) => ({ id, data })
    );
    const exitUpdate = pendingExitUpdates.current;

    if (assumptionUpdates.length > 0) {
      bulkUpdateMutation.mutate(assumptionUpdates);
      pendingAssumptionUpdates.current.clear();
    }
    if (exitUpdate) {
      exitUpdateMutation.mutate(exitUpdate);
      pendingExitUpdates.current = null;
    }
  }, [bulkUpdateMutation, exitUpdateMutation]);

  const scheduleSave = useCallback(() => {
    setSaveStatus('saving');
    if (debounceRef.current) clearTimeout(debounceRef.current);
    debounceRef.current = setTimeout(flushSave, 800);
  }, [flushSave]);

  useEffect(() => {
    return () => {
      if (debounceRef.current) clearTimeout(debounceRef.current);
    };
  }, []);

  // ─── Update handlers ───
  const updateAssumption = useCallback((assumptionId: number, field: string, value: number) => {
    // Update local state immediately for instant UI
    setLocalAssumptions(prev => ({
      ...prev,
      [assumptionId]: { ...(prev[assumptionId] || {}), [field]: value },
    }));

    // Queue for save
    const existing = pendingAssumptionUpdates.current.get(assumptionId) || {};
    pendingAssumptionUpdates.current.set(assumptionId, { ...existing, [field]: String(value) });
    scheduleSave();
  }, [scheduleSave]);

  // In "simple" mode, update ALL years at once
  const updateAllYears = useCallback((field: string, value: number) => {
    for (const a of sorted) {
      updateAssumption(a.id, field, value);
    }
  }, [sorted, updateAssumption]);

  const updateExit = useCallback((field: string, value: number) => {
    setLocalExit(prev => prev ? { ...prev, [field]: value } : null);
    pendingExitUpdates.current = { ...(pendingExitUpdates.current || {}), [field]: String(value) };
    scheduleSave();
  }, [scheduleSave]);

  // ─── Helpers ───
  const getLocal = (assumptionId: number, field: string): number => {
    return localAssumptions[assumptionId]?.[field] ?? 0;
  };

  // For simple mode, use Year 1 values
  const firstAssumption = sorted[0];
  const simpleVal = (field: string): number => {
    if (!firstAssumption) return 0;
    return getLocal(firstAssumption.id, field);
  };

  // Funding summary
  const totalDebt = fundingLines
    .filter(f => f.isDebt && f.isActive)
    .reduce((sum, f) => sum + toNum(f.amountUsed), 0);
  const avgRate = (() => {
    const debtLines = fundingLines.filter(f => f.isDebt && f.isActive && toNum(f.amountUsed) > 0);
    if (debtLines.length === 0) return 0;
    const weighted = debtLines.reduce((sum, f) => sum + toNum(f.interestRate) * toNum(f.amountUsed), 0);
    return weighted / debtLines.reduce((sum, f) => sum + toNum(f.amountUsed), 0);
  })();

  return (
    <div className="space-y-3">
      {/* Save indicator */}
      <div className="flex items-center justify-between">
        <h3 className="text-sm font-semibold text-muted-foreground uppercase tracking-wide">Assumptions</h3>
        {saveStatus !== 'idle' && (
          <span className="flex items-center gap-1 text-xs text-muted-foreground">
            {saveStatus === 'saving' ? (
              <><Loader2 className="h-3 w-3 animate-spin" /> Saving</>
            ) : (
              <><Check className="h-3 w-3 text-green-600" /> Saved</>
            )}
          </span>
        )}
      </div>

      {/* Scenario label */}
      <Badge variant="secondary" className="text-xs">
        {scenario.name} — {scenario.scenarioType === 'bear' ? '🐻' : scenario.scenarioType === 'bull' ? '🐂' : '📊'}
      </Badge>

      {/* ═══ Growth Assumptions ═══ */}
      <Collapsible open={growthOpen} onOpenChange={setGrowthOpen}>
        <CollapsibleTrigger className="flex items-center gap-2 w-full py-2 hover:bg-muted/50 rounded-md px-2 transition-colors">
          {growthOpen ? <ChevronDown className="h-4 w-4" /> : <ChevronRight className="h-4 w-4" />}
          <TrendingUp className="h-4 w-4 text-blue-600" />
          <span className="text-sm font-medium flex-1 text-left">Growth Assumptions</span>
        </CollapsibleTrigger>
        <CollapsibleContent className="space-y-3 pt-2">
          {/* Simple / Per Year toggle */}
          <div className="flex items-center gap-2 px-2">
            <Label htmlFor="per-year-toggle" className="text-xs text-muted-foreground">
              {perYearMode ? 'Per year' : 'Uniform'}
            </Label>
            <Switch
              id="per-year-toggle"
              checked={perYearMode}
              onCheckedChange={setPerYearMode}
            />
          </div>

          {perYearMode ? (
            // ─── Per-Year Mode ───
            <div className="space-y-4">
              {sorted.map(a => (
                <div key={a.id} className="space-y-2 border rounded-lg p-2 bg-muted/20">
                  <p className="text-xs font-semibold text-muted-foreground">Year {a.year}</p>
                  
                  <div className="space-y-1.5">
                    <div className="flex items-center justify-between">
                      <span className="text-xs">Revenue growth</span>
                      <NumericStepper
                        value={getLocal(a.id, 'revenueGrowthPercent')}
                        onChange={v => updateAssumption(a.id, 'revenueGrowthPercent', v)}
                      />
                    </div>
                    <div className="flex items-center justify-between">
                      <span className="text-xs">CoS inflation</span>
                      <NumericStepper
                        value={getLocal(a.id, 'costOfSalesGrowthPercent')}
                        onChange={v => updateAssumption(a.id, 'costOfSalesGrowthPercent', v)}
                      />
                    </div>
                    <div className="flex items-center justify-between">
                      <span className="text-xs">Staff cost growth</span>
                      <NumericStepper
                        value={getLocal(a.id, 'staffCostGrowthPercent')}
                        onChange={v => updateAssumption(a.id, 'staffCostGrowthPercent', v)}
                      />
                    </div>
                    <div className="flex items-center justify-between">
                      <span className="text-xs">Admin cost growth</span>
                      <NumericStepper
                        value={getLocal(a.id, 'adminCostGrowthPercent')}
                        onChange={v => updateAssumption(a.id, 'adminCostGrowthPercent', v)}
                      />
                    </div>
                    <div className="flex items-center justify-between">
                      <span className="text-xs">Capex</span>
                      <CurrencyStepper
                        value={getLocal(a.id, 'capexPerYear')}
                        onChange={v => updateAssumption(a.id, 'capexPerYear', v)}
                      />
                    </div>
                    <div className="flex items-center justify-between">
                      <span className="text-xs">Tax rate</span>
                      <NumericStepper
                        value={getLocal(a.id, 'taxRatePercent')}
                        onChange={v => updateAssumption(a.id, 'taxRatePercent', v)}
                        step={1}
                        min={0}
                        max={50}
                      />
                    </div>
                  </div>
                </div>
              ))}
            </div>
          ) : (
            // ─── Simple (Uniform) Mode ───
            <div className="space-y-2 px-1">
              <div className="flex items-center justify-between">
                <span className="text-xs">Revenue growth</span>
                <NumericStepper
                  value={simpleVal('revenueGrowthPercent')}
                  onChange={v => updateAllYears('revenueGrowthPercent', v)}
                />
              </div>
              <div className="flex items-center justify-between">
                <span className="text-xs">CoS inflation</span>
                <NumericStepper
                  value={simpleVal('costOfSalesGrowthPercent')}
                  onChange={v => updateAllYears('costOfSalesGrowthPercent', v)}
                />
              </div>
              <div className="flex items-center justify-between">
                <span className="text-xs">Staff cost growth</span>
                <NumericStepper
                  value={simpleVal('staffCostGrowthPercent')}
                  onChange={v => updateAllYears('staffCostGrowthPercent', v)}
                />
              </div>
              <div className="flex items-center justify-between">
                <span className="text-xs">Admin cost growth</span>

        </>
      )}
    </div>
  );
}
```
