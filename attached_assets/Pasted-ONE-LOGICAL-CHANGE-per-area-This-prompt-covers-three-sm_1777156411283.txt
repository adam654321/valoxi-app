PART A — RESEND INVITATION REGENERATION
================================================================

The current PUT /api/v2/team/invitations/:id handler accepts a free-form
status string and only flips invitation_status. The frontend uses this
for both Cancel ({status:'cancelled'}) and Resend ({status:'pending'}).
Resend currently does NOT regenerate the token, does NOT re-send the
email, and does NOT extend expiresAt.

Replace the single status-mutation endpoint with two distinct purpose-
built endpoints, then update the frontend to call them.

Replace the existing PUT /invitations/:id handler with:

A2. server/services/team.service.ts
Delete the existing updateInvitation method (it is no longer needed).

Add two new methods:

  async resendInvitation(
    invitationId: number,
    userId: number,
    userCompanyId: number,
    baseUrl: string,
    getUserWithPermissions?: (id: number) => Promise<UserWithPermissions | undefined>
  ): Promise<UserInvitation> {
    if (getUserWithPermissions) {
      const userPerms = await getUserWithPermissions(userId);
      if (!userPerms?.permissions?.canManageUsers && !userPerms?.isCompanyAdmin) {
        throw new ForbiddenError('Insufficient permissions');
      }
    }
    const existing = await teamRepository.getInvitationByIdAndCompany(invitationId, userCompanyId);
    if (!existing) throw new NotFoundError('Invitation not found');
    if (existing.invitationStatus !== 'pending') {
      throw new BadRequestError('Only pending invitations can be resent');
    }

    // Rotate token and extend expiry. Status remains 'pending'.
    const newToken = randomUUID();
    const newExpiresAt = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000);
    const updated = await teamRepository.updateInvitation(invitationId, {
      invitationToken: newToken,
      expiresAt: newExpiresAt,
    });
    if (!updated) throw new NotFoundError('Invitation not found');

    // Re-send the email (fire-and-forget, matches createInvitation pattern)
    emailApiService.sendTeamInvitation(invitationId, baseUrl).catch(err => {
      console.error(`[TEAM] Failed to re-send invitation email for invitation ${invitationId}:`, err.message);
    });

    return updated;
  }

  async cancelInvitation(
    invitationId: number,
    userId: number,
    userCompanyId: number,
    getUserWithPermissions?: (id: number) => Promise<UserWithPermissions | undefined>
  ): Promise<UserInvitation> {
    if (getUserWithPermissions) {
      const userPerms = await getUserWithPermissions(userId);
      if (!userPerms?.permissions?.canManageUsers && !userPerms?.isCompanyAdmin) {
        throw new ForbiddenError('Insufficient permissions');
      }
    }
    const existing = await teamRepository.getInvitationByIdAndCompany(invitationId, userCompanyId);
    if (!existing) throw new NotFoundError('Invitation not found');

    // Cancel the invitation AND deactivate the pre-created shell user.
    // The shell was created with isActive=true at invite time; on cancel
    // it must be deactivated so it does not consume a seat or appear in
    // the team list. Cancellation is reversible: if the inviter changes
    // their mind they send a fresh invite, which goes through the
    // cross-company guard check and (if the shell still exists) errors
    // out with "User already exists in your company" — at which point
    // the inviter can re-activate manually or delete-and-reinvite.
    // For now we accept that operational cost; the invitee email is
    // unique across the platform so we cannot just leave a deactivated
    // shell sitting there forever.
    const cancelled = await teamRepository.cancelInvitationAndDeactivateShell(
      invitationId, userCompanyId, existing.email
    );
    if (!cancelled) throw new NotFoundError('Invitation not found');
    return cancelled;
  }

A3. server/repositories/team.repository.ts
Add helper methods:

  async getInvitationByIdAndCompany(
    id: number, userCompanyId: number
  ): Promise<UserInvitation | undefined> {
    const [row] = await db.select().from(userInvitations)
      .where(and(eq(userInvitations.id, id), eq(userInvitations.userCompanyId, userCompanyId)))
      .limit(1);
    return row || undefined;
  }

  async cancelInvitationAndDeactivateShell(
    invitationId: number, userCompanyId: number, email: string
  ): Promise<UserInvitation | undefined> {
    return await db.transaction(async (tx) => {
      // Cancel the invitation
      const [cancelled] = await tx.update(userInvitations)
        .set({ invitationStatus: 'cancelled' })
        .where(and(
          eq(userInvitations.id, invitationId),
          eq(userInvitations.userCompanyId, userCompanyId)
        ))
        .returning();
      if (!cancelled) return undefined;

      // Deactivate the shell — only if it is still unlinked
      // (clerk_user_id IS NULL). If the user has linked since invite,
      // do NOT deactivate them: that would deactivate a real user.
      await tx.update(users)
        .set({ isActive: false })
        .where(and(
          sql`LOWER(${users.email}) = LOWER(${email})`,
          eq(users.userCompanyId, userCompanyId),
          sql`${users.clerkUserId} IS NULL`
        ));

      return cancelled;
    });
  }
