const getCompanyProfile = async (companyId, userId) => {
  // Verify access and get complete company data
  const query = `
    SELECT 
      t.*,
      u.first_name || ' ' || u.last_name as created_by_name,
      uc.company_name as user_company_name,
      COUNT(tc.id) as total_contacts,
      primary_contact.first_name || ' ' || primary_contact.last_name as primary_contact_name,
      primary_contact.email as primary_contact_email,
      primary_contact.phone as primary_contact_phone,
      primary_contact.title as primary_contact_title
    FROM target_companies t
    JOIN users u ON t.created_by = u.id
    JOIN user_companies uc ON t.user_company_id = uc.id
    LEFT JOIN target_contacts tc ON t.id = tc.target_company_id
    LEFT JOIN target_contacts primary_contact ON t.id = primary_contact.target_company_id AND primary_contact.is_primary = true
    WHERE t.id = $1 AND t.user_company_id = (
      SELECT user_company_id FROM users WHERE id = $2
    )
    GROUP BY t.id, u.first_name, u.last_name, uc.company_name, 
             primary_contact.first_name, primary_contact.last_name, 
             primary_contact.email, primary_contact.phone, primary_contact.title
  `;
  
  const result = await db.query(query, [companyId, userId]);
  
  if (result.rows.length === 0) {
    throw new Error('Company not found or access denied');
  }
  
  const company = result.rows[0];
  
  // Calculate additional metrics
  company.years_in_business = company.year_established ? 
    new Date().getFullYear() - company.year_established : null;
  
  company.revenue_per_employee = (company.annual_revenue && company.employee_count) ?
    company.annual_revenue / company.employee_count : null;
  
  // Get all contacts
  const contacts = await db.query(
    'SELECT * FROM target_contacts WHERE target_company_id = $1 ORDER BY is_primary DESC, last_name ASC',
    [companyId]
  );
  
  company.contacts = contacts.rows;
  
  return company;
};

// Quick action functions
const emailContact = (email) => {
  if (email) {
    window.location.href = `mailto:${email}`;
  }
};

const callContact = (phone) => {
  if (phone) {
    window.location.href = `tel:${phone}`;
  }
};

const editCompany = (companyId) => {
  window.location.href = `/edit-target-company.html?id=${companyId}`;
};

const addContact = (companyId) => {
  window.location.href = `/add-contact.html?company=${companyId}`;
};

// Format currency values
const formatCurrency = (amount) => {
  if (!amount) return 'Not specified';
  return new Intl.NumberFormat('en-US', {
    style: 'currency',
    currency: 'USD',
    minimumFractionDigits: 0,
    maximumFractionDigits: 0
  }).format(amount);
};

// Format dates
const formatDate = (dateString) => {
  if (!dateString) return 'Not specified';
  return new Date(dateString).toLocaleDateString('en-US', {
    year: 'numeric',
    month: 'long',
    day: 'numeric'
  });
};
