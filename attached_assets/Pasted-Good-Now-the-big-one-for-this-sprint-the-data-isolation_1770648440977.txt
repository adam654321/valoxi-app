2. The query should filter on both targetCompanies.id === targetCompanyId AND targetCompanies.userCompanyId === userCompanyId
3. If no result is found, throw NotFoundError (this is correct — we don't want to reveal whether the company exists to an unauthorised user, just say not found)

async function verifyOwnership(targetCompanyId: number, userCompanyId: number): Promise<void> {
  const [company] = await db
    .select({ id: targetCompanies.id })
    .from(targetCompanies)
    .where(
      and(
        eq(targetCompanies.id, targetCompanyId),
        eq(targetCompanies.userCompanyId, userCompanyId)
      )
    );
  if (!company) {
    throw new NotFoundError('Target company not found');
  }
}
