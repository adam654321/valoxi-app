```typescript
router.get('/:token/dd-requirements', async (req, res) => {
  try {
    const { token } = req.params;
    
    // Validate portal token (same pattern as other portal endpoints)
    // Get the targetCompanyId and userCompanyId from the validated token
    // Use whatever method the other GET endpoints use to validate and extract these

    // Get DD categories with requirements
    const categories = await dueDiligenceService.getAllCategories();
    
    // Get DD status for this company
    const status = await dueDiligenceService.getDdStatusForCompany(targetCompanyId, userCompanyId);
    
    // Get completion summary
    const summary = await dueDiligenceService.getCompletionSummary(targetCompanyId, userCompanyId);

    // Merge requirements with their status
    // For each category, for each requirement, find the matching status entry
    const categoriesWithStatus = categories.map((cat: any) => ({
      ...cat,
      requirements: (cat.requirements || []).map((req: any) => {
        const reqStatus = status.find((s: any) => s.requirementId === req.id);
        return {
          id: req.id,
          refNumber: req.refNumber,
          documentName: req.documentName,
          description: req.description,
          priority: req.priority,
          category: cat.categoryName,
          status: reqStatus?.status || 'not_started',
          documentId: reqStatus?.documentId || null,
          externalSubmissionId: reqStatus?.externalSubmissionId || null,
        };
      }),
    }));

    res.json({
      categories: categoriesWithStatus,
      summary,
    });
  } catch (error: any) {
    console.error('[PORTAL] DD requirements error:', error);
    res.status(500).json({ message: error.message });
  }
});
```
