STEP 2 — Create server/services/deal-document-template.service.ts:
```typescript
import { createReport } from 'docx-templates';
import fs from 'fs/promises';
import path from 'path';
import { db } from '../db';
import { dealDocumentTemplates, dealDocuments, dealDocumentResponses } from '@shared/schema/deal-documents';
import { targetCompanies } from '@shared/schema/deals';
import { userCompanies, users } from '@shared/schema/core';
import { dealModels } from '@shared/schema/deal-models';
import { eq, and, desc } from 'drizzle-orm';

// Directory for storing generated documents and templates
const TEMPLATES_DIR = path.join(process.cwd(), 'data', 'templates');
const GENERATED_DIR = path.join(process.cwd(), 'data', 'generated-documents');

export interface DocumentDataContext {
  // Buyer details
  buyerCompanyName: string;
  buyerAddress?: string;
  buyerEmail?: string;
  buyerPhone?: string;
  buyerSignatoryName: string;
  buyerSignatoryTitle?: string;
  
  // Seller/Target details
  sellerCompanyName: string;
  sellerCompanyNumber?: string;
  sellerAddress?: string;
  sellerContactName?: string;
  sellerContactEmail?: string;
  
  // Date
  currentDate: string; // formatted as DD/MM/YYYY
  
  // Deal Modeller values (populated for offers and HoTs)
  averagingYears?: number;
  averageEbitda?: number;
  ebitdaMultiple?: number;
  enterpriseValue?: number;
  offerPrice?: number;
  propertyValue?: number;
  excessCash?: number;
  primaryDebt?: number;
  otherDebt?: number;
  corpTaxDue?: number;
  workingCapitalAdj?: number;
  totalFees?: number;
  totalFundingRequired?: number;
  
  // Payment structure
  dayOnePayment?: number;
  deferredTotal?: number;
  deferredSchedule?: Array<{
    period: string;
    amount: number;
  }>;
  
  // Funding lines
  fundingLines?: Array<{
    name: string;
    amount: number;
    type: string;
    interestRate?: number;
    termMonths?: number;
  }>;
  
  // EBITDA by year (for showing the calculation)
  ebitdaByYear?: Record<string, number>;
  
  // Fees breakdown
  feeLegalDd?: number;
  feeAccountingDd?: number;
  feeBroker?: number;
  feeLoanArrangement?: number;
  feeDealMaker?: number;
  
  // HoT specific
  exclusivityPeriodDays?: number;
  nonCompetePeriodYears?: number;
  nonSolicitPeriodYears?: number;
  warrantyClaimPeriodMonths?: number;
  warrantyMinClaim?: number;
  warrantyAggregateCap?: number;
  
  // Custom fields
  [key: string]: any;
}

export class DealDocumentTemplateService {
  
  /**
   * Ensure storage directories exist
   */
  async ensureDirectories(): Promise<void> {
    await fs.mkdir(TEMPLATES_DIR, { recursive: true });
    await fs.mkdir(GENERATED_DIR, { recursive: true });
  }
  
  /**
   * Generate a populated document from a template
   */
  async generateDocument(
    templatePath: string,
    data: DocumentDataContext
  ): Promise<Buffer> {
    const templateBuffer = await fs.readFile(templatePath);
    
    const buffer = await createReport({
      template: templateBuffer,
      data,
      cmdDelimiter: ['{{', '}}'],
      processLineBreaks: true,
      failFast: false, // Don't crash on missing placeholders — leave them blank
    });
    
    return Buffer.from(buffer);
  }
  
  /**
   * Build the data context for a document from the database
   */
  async buildDataContext(
    targetCompanyId: number,
    userCompanyId: number,
    createdBy: number,
    dealModelId?: number | null
  ): Promise<DocumentDataContext> {
    // Fetch buyer company
    const [buyerCompany] = await db
      .select()
      .from(userCompanies)
      .where(eq(userCompanies.id, userCompanyId));
    
    // Fetch buyer user (signatory)
    const [buyerUser] = await db
      .select()
      .from(users)
      .where(eq(users.id, createdBy));
    
    // Fetch target company
    const [targetCompany] = await db
      .select()
      .from(targetCompanies)
      .where(eq(targetCompanies.id, targetCompanyId));
    
    // Build base context
    const context: DocumentDataContext = {
      buyerCompanyName: buyerCompany?.companyName || '',
      buyerAddress: buyerCompany?.address || '',
      buyerEmail: buyerCompany?.email || '',
      buyerPhone: buyerCompany?.phone || '',
      buyerSignatoryName: buyerUser ? `${buyerUser.firstName} ${buyerUser.lastName}` : '',
      
      sellerCompanyName: targetCompany?.companyName || '',
      sellerCompanyNumber: (targetCompany as any)?.companyNumber || '',
      sellerAddress: (targetCompany as any)?.address || '',
      
      currentDate: new Date().toLocaleDateString('en-GB', {
        day: '2-digit',
        month: '2-digit', 
        year: 'numeric',
      }),
    };
    
    // If we have a deal model, add financial data
    if (dealModelId) {
      const [model] = await db
        .select()
        .from(dealModels)
        .where(eq(dealModels.id, dealModelId));
      
      if (model) {
        // Import calculator to get computed values
        const { calculateFullDealModel } = await import('@shared/calculators/deal-model.calc');
        
        // We need the full calculation — but buildCalcInputs is in the service
        // For now, store the raw model values; the caller can pass in pre-calculated values
        context.ebitdaMultiple = Number(model.ebitdaMultiple) || 0;
        context.averagingYears = model.averagingYears || 3;
        context.propertyValue = Number(model.plusPropertyValue || model.propertyValue) || 0;
        context.excessCash = Number(model.plusExcessCash) || 0;
        context.primaryDebt = Number(model.minusPrimaryDebt) || 0;
        context.otherDebt = Number(model.minusOtherDebt) || 0;
        context.corpTaxDue = Number(model.minusCorpTaxDue) || 0;
        context.workingCapitalAdj = Number(model.minusWorkingCapitalAdj) || 0;
      }
    }
    
    return context;
  }
  
  /**
   * Generate and save a document, returning the file path
   */
  async generateAndSave(
    templatePath: string,
    data: DocumentDataContext,
    filename: string
  ): Promise<string> {
    await this.ensureDirectories();
    
    const buffer = await this.generateDocument(templatePath, data);
    const outputPath = path.join(GENERATED_DIR, filename);
    await fs.writeFile(outputPath, buffer);
    
    return outputPath;
  }
  
  /**
   * Get a system template by document type
   */
  async getSystemTemplate(documentType: string): Promise<typeof dealDocumentTemplates.$inferSelect | null> {
    const [template] = await db
      .select()
      .from(dealDocumentTemplates)
      .where(
        and(
          eq(dealDocumentTemplates.documentType, documentType),
          eq(dealDocumentTemplates.isSystem, true),
          eq(dealDocumentTemplates.isActive, true)
        )
      );
    return template || null;
  }
  
  /**
   * Format a number as GBP currency string
   */
  formatCurrency(amount: number): string {
    return new Intl.NumberFormat('en-GB', {
      style: 'currency',
      currency: 'GBP',
      minimumFractionDigits: 0,
      maximumFractionDigits: 0,
    }).format(amount);
  }
  
  /**
   * Format a number as GBP with pence
   */
  formatCurrencyPrecise(amount: number): string {
    return new Intl.NumberFormat('en-GB', {
      style: 'currency',
      currency: 'GBP',
      minimumFractionDigits: 2,
      maximumFractionDigits: 2,
    }).format(amount);
  }
}

export const dealDocumentTemplateService = new DealDocumentTemplateService();
```

IMPORTANT:
- The cmdDelimiter ['{{', '}}'] means template placeholders look like {{buyerCompanyName}}, {{offerPrice}}, etc.
- failFast: false means missing placeholders render as empty rather than crashing
- processLineBreaks: true preserves line breaks in the generated output
- The DocumentDataContext interface is deliberately broad — different document types will use different subsets
- The buildDataContext method reads from the database but can be supplemented with pre-calculated values by the caller
