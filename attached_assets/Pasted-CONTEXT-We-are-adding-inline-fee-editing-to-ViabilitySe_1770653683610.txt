  return (field: string, value: any) => {
    pendingRef.current[field] = value;
    clearTimeout(timer);
    timer = setTimeout(() => {
      const updates = { ...pendingRef.current };
      pendingRef.current = {};
      setSaveStatus('saving');
      updateMutation.mutate(updates);
    }, 600);
  };
}, [modelId]);
