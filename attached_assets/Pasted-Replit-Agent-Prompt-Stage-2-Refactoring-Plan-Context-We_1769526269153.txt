Update server/storage.ts:

Import all the new Store classes.
Update the DatabaseStorage class to initialize these stores.
Crucial: Keep the IStorage interface definition in this file (or move to types.ts and re-export) so imports don't break.
Refactor DatabaseStorage methods to delegate to the specific stores.
Example Pattern:

// In server/storage.ts
export class DatabaseStorage implements IStorage {
  private userStore: UserStore;
  private companyStore: CompanyStore;
  // ...
  constructor() {
    this.userStore = new UserStore(db);
    this.companyStore = new CompanyStore(db);
    // ...
  }
  // Delegate
  async getUser(id: number) {
    return this.userStore.getUser(id);
  }
  
  async getTargetCompanies(...) {
    return this.companyStore.getTargetCompanies(...);
  }
}
