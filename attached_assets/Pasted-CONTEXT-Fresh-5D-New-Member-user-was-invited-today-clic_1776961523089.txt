  import Stripe from 'stripe';
  import { config } from '../config';
  
  const stripe = new Stripe(config.stripe.secretKey!);
  
  const SUBSCRIPTION_ID = '<paste stripe_subscription_id from CHECK 2>';
  const CUSTOMER_ID = '<paste stripe_customer_id from CHECK 2>';
  
  async function main() {
    // 1. Fetch the subscription — does it have trial_end/trial_start?
    const sub = await stripe.subscriptions.retrieve(SUBSCRIPTION_ID);
    console.log('SUB trial_start:', sub.trial_start);
    console.log('SUB trial_end:', sub.trial_end);
    console.log('SUB status:', sub.status);
    console.log('SUB metadata:', JSON.stringify(sub.metadata));
    console.log('SUB created:', new Date(sub.created * 1000).toISOString());
    
    // 2. Find the most recent checkout session for this customer
    const sessions = await stripe.checkout.sessions.list({ 
      customer: CUSTOMER_ID, 
      limit: 3 
    });
    for (const s of sessions.data) {
      console.log('---');
      console.log('SESSION id:', s.id, 'created:', new Date(s.created * 1000).toISOString());
      console.log('SESSION mode:', s.mode, 'status:', s.status);
      // subscription_data is not on the session response by default; we need to look at the sub link
      console.log('SESSION subscription:', s.subscription);
      console.log('SESSION amount_total:', s.amount_total);
      console.log('SESSION metadata:', JSON.stringify(s.metadata));
    }
    
    // 3. Fetch the Pro Monthly price — does it have a default trial_period_days?
    const prices = await stripe.prices.list({ 
      lookup_keys: ['ma_professional_monthly_usd'], 
      active: true, 
      limit: 1,
      expand: ['data.product']
    });
    if (prices.data.length) {
      const p = prices.data[0];
      console.log('---');
      console.log('PRICE id:', p.id);
      console.log('PRICE recurring:', JSON.stringify(p.recurring));
      console.log('PRICE product:', (p.product as any).id, 'name:', (p.product as any).name);
      console.log('PRICE product metadata:', JSON.stringify((p.product as any).metadata));
    }
    
    // 4. Fetch the customer — do they have any default trial config?
    const customer = await stripe.customers.retrieve(CUSTOMER_ID);
    if (customer && !customer.deleted) {
      console.log('---');
      console.log('CUSTOMER balance:', (customer as any).balance);
      console.log('CUSTOMER metadata:', JSON.stringify((customer as any).metadata));
    }
  }
  
  main().catch(e => { console.error(e); process.exit(1); });
