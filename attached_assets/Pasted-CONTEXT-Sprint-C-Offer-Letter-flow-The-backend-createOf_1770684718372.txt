```typescript
// POST /api/v2/deal-documents/send-offer
// Create and send an offer letter from a deal model
router.post('/send-offer', async (req, res) => {
  try {
    const userId = req.session.userId;
    const userCompanyId = req.session.userCompanyId;
    if (!userId || !userCompanyId) {
      return res.status(401).json({ message: 'Unauthorized' });
    }
    
    const { targetCompanyId, dealModelId, portalTokenId } = req.body;
    
    if (!targetCompanyId || !dealModelId) {
      return res.status(400).json({ message: 'targetCompanyId and dealModelId are required' });
    }
    
    // If no portalTokenId provided, look up the active portal for this company
    let resolvedPortalTokenId = portalTokenId;
    if (!resolvedPortalTokenId) {
      const { portalService } = await import('../services/portal.service');
      // Look up active portal token for this target company
      const activeToken = await portalService.getActiveTokenForCompany(targetCompanyId);
      if (!activeToken) {
        return res.status(400).json({ 
          message: 'No active portal found for this company. Please create a portal first.' 
        });
      }
      resolvedPortalTokenId = activeToken.id;
    }
    
    const doc = await dealDocumentService.createOfferFromDealModel({
      targetCompanyId,
      userCompanyId,
      dealModelId,
      portalTokenId: resolvedPortalTokenId,
      createdBy: userId,
    });
    
    // Auto-send the offer
    const sent = await dealDocumentService.sendDocument(doc.id);
    
    res.status(201).json(sent);
  } catch (error: any) {
    console.error('[DEAL_DOCS] Send offer error:', error);
    res.status(500).json({ message: error.message });
  }
});
```

NOTE: The route uses portalService.getActiveTokenForCompany() — this method may not exist yet. Check portal.service.ts for a method that finds the active portal token for a given targetCompanyId. If it doesn't exist, add one:
```typescript
  /**
   * Get the active portal token for a target company
   */
  async getActiveTokenForCompany(targetCompanyId: number) {
    const [token] = await db.select().from(portalAccessTokens)
      .where(
        and(
          eq(portalAccessTokens.targetCompanyId, targetCompanyId),
          eq(portalAccessTokens.isActive, true)
        )
      )
      .orderBy(desc(portalAccessTokens.createdAt))
      .limit(1);
    return token || null;
  }
```
