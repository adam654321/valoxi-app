1. server/services/file-storage.service.ts — wrong bucket ID and swallowed errors
The Client() is constructed with no arguments so it connects to the wrong bucket. Fix the getClient() function to extract the bucket ID from the PRIVATE_OBJECT_DIR environment variable, which contains it in the path:
typescriptfunction getClient(): Client {
  if (!clientInstance) {
    const privateDir = process.env.PRIVATE_OBJECT_DIR ?? '';
    const bucketId = privateDir.split('/')[1] ?? undefined;
    clientInstance = bucketId ? new Client({ bucketId }) : new Client();
    console.log(`[FILE_STORAGE] Initialised object storage client, bucketId: ${bucketId ?? 'default'}`);
  }
  return clientInstance;
}
