Replit Development Prompt 5a: Team Member Invitations & User Management
Prerequisites
Prompts 1-5 completed with working target company profiles and contact management
User authentication system operational with company data isolation
Basic user and company management functionality working
What This Prompt Builds
Add comprehensive team collaboration functionality allowing users to invite colleagues from their company to join the M&A CRM system, manage team member permissions, and enable collaborative deal management.
System Context Reminder
Team Invitations: Allow existing users to invite colleagues to join their company account
User Roles: Different permission levels for team members (Admin, Manager, Analyst)
Collaborative Access: Multiple team members can work on the same deals with proper access control
What You Need to Build
1. Team Management System
Create comprehensive team collaboration with these capabilities:

User Invitation Process:

Send email invitations to new team members
Secure invitation links with expiration
Role assignment during invitation
Invitation tracking and management

User Roles and Permissions:

Company Admin - Full system access, user management, billing
Deal Manager - Manage deals, invite users, view all company data
Senior Analyst - Create/edit deals, upload documents, create offers
Analyst - View deals, add activities, upload documents (limited)
Viewer - Read-only access to assigned deals

Team Collaboration:

Assign team members to specific deals
Activity notifications between team members
Deal ownership and responsibility tracking
2. Database Schema for Team Management
-- Create user invitations table
CREATE TABLE user_invitations (
  id SERIAL PRIMARY KEY,
  user_company_id INTEGER REFERENCES user_companies(id),
  email VARCHAR(255) NOT NULL,
  invited_by INTEGER REFERENCES users(id),
  
  -- Invitation Details
  invitation_token VARCHAR(255) UNIQUE NOT NULL,
  invited_role VARCHAR(50) NOT NULL, -- company_admin, deal_manager, senior_analyst, analyst, viewer
  invitation_message TEXT,
  
  -- Status Tracking
  invitation_status VARCHAR(50) DEFAULT 'pending', -- pending, accepted, expired, cancelled
  sent_at TIMESTAMP DEFAULT NOW(),
  expires_at TIMESTAMP DEFAULT (NOW() + INTERVAL '7 days'),
  accepted_at TIMESTAMP,
  
  -- Security
  ip_address_sent INET,
  ip_address_accepted INET,
  
  created_at TIMESTAMP DEFAULT NOW()
);

-- Create user roles table
CREATE TABLE user_roles (
  id SERIAL PRIMARY KEY,
  role_name VARCHAR(50) UNIQUE NOT NULL,
  role_description TEXT,
  
  -- Permissions
  can_manage_users BOOLEAN DEFAULT FALSE,
  can_manage_company BOOLEAN DEFAULT FALSE,
  can_create_deals BOOLEAN DEFAULT FALSE,
  can_edit_all_deals BOOLEAN DEFAULT FALSE,
  can_delete_deals BOOLEAN DEFAULT FALSE,
  can_upload_documents BOOLEAN DEFAULT FALSE,
  can_create_offers BOOLEAN DEFAULT FALSE,
  can_view_financials BOOLEAN DEFAULT FALSE,
  can_access_analytics BOOLEAN DEFAULT FALSE,
  can_export_data BOOLEAN DEFAULT FALSE,
  
  created_at TIMESTAMP DEFAULT NOW()
);

-- Create user assignments table (many-to-many for users and target companies)
CREATE TABLE user_deal_assignments (
  id SERIAL PRIMARY KEY,
  user_id INTEGER REFERENCES users(id),
  target_company_id INTEGER REFERENCES target_companies(id),
  assigned_by INTEGER REFERENCES users(id),
  
  -- Assignment Details
  assignment_role VARCHAR(50) DEFAULT 'analyst', -- lead, analyst, viewer
  assignment_date TIMESTAMP DEFAULT NOW(),
  is_active BOOLEAN DEFAULT TRUE,
  
  -- Permissions for this specific deal
  can_edit_company BOOLEAN DEFAULT TRUE,
  can_upload_documents BOOLEAN DEFAULT TRUE,
  can_create_activities BOOLEAN DEFAULT TRUE,
  can_manage_contacts BOOLEAN DEFAULT TRUE,
  
  UNIQUE(user_id, target_company_id)
);

4. Team Management Implementation
const teamManager = {
  // Send invitation to new team member
  inviteTeamMember: async (inviterUserId, invitationData) => {
    // Verify inviter has permission to invite users
    const inviter = await getUserWithPermissions(inviterUserId);
    if (!inviter.can_manage_users && !inviter.is_company_admin) {
      throw new Error('Insufficient permissions to invite users');
    }
    
    // Check if user already exists or has pending invitation
    const existingUser = await db.query(
      'SELECT id FROM users WHERE email = $1 AND user_company_id = $2',
      [invitationData.email, inviter.user_company_id]
    );
    
    if (existingUser.rows.length > 0) {
      throw new Error('User already exists in your company');
    }
    
    const existingInvitation = await db.query(
      'SELECT id FROM user_invitations WHERE email = $1 AND user_company_id = $2 AND invitation_status = $3',
      [invitationData.email, inviter.user_company_id, 'pending']
    );
    
    if (existingInvitation.rows.length > 0) {
      throw new Error('Invitation already sent to this email address');
    }
    
    // Generate secure invitation token
    const crypto = require('crypto');
    const invitationToken = crypto.randomBytes(32).toString('hex');
    
    // Create invitation record
    const invitationQuery = `
      INSERT INTO user_invitations (
        user_company_id, email, invited_by, invitation_token, invited_role, invitation_message
      ) VALUES ($1, $2, $3, $4, $5, $6)
      RETURNING *
    `;
    
    const invitationValues = [
      inviter.user_company_id,
      invitationData.email.toLowerCase(),
      inviterUserId,
      invitationToken,
      invitationData.role || 'analyst',
      invitationData.message || ''
    ];
    
    const result = await db.query(invitationQuery, invitationValues);
    const invitation = result.rows[0];
    
    // Send invitation email
    await sendInvitationEmail(invitation, inviter);
    
    // Log activity
    await logActivity(null, inviterUserId, 'team_invitation_sent', `Invited ${invitationData.email} as ${invitationData.role}`);
    
    return invitation;
  },
  
  // Accept invitation and create user account
  acceptInvitation: async (invitationToken, userData) => {
    // Verify invitation token
    const invitationQuery = `
      SELECT ui.*, uc.company_name 
      FROM user_invitations ui
      JOIN user_companies uc ON ui.user_company_id = uc.id
      WHERE ui.invitation_token = $1 
      AND ui.invitation_status = 'pending'
      AND ui.expires_at > NOW()
    `;
    
    const invitationResult = await db.query(invitationQuery, [invitationToken]);
    
    if (invitationResult.rows.length === 0) {
      throw new Error('Invalid or expired invitation');
    }
    
    const invitation = invitationResult.rows[0];
    
    // Create user account
    const hashedPassword = await bcrypt.hash(userData.password, 10);
    
    const userQuery = `
      INSERT INTO users (
        email, password_hash, first_name, last_name, user_company_id, user_role, is_active
      ) VALUES ($1, $2, $3, $4, $5, $6, $7)
      RETURNING *
    `;
    
    const userValues = [
      invitation.email,
      hashedPassword,
      userData.first_name,
      userData.last_name,
      invitation.user_company_id,
      invitation.invited_role,
      true
    ];
    
    const userResult = await db.query(userQuery, userValues);
    const newUser = userResult.rows[0];
    
    // Update invitation status
    await db.query(
      'UPDATE user_invitations SET invitation_status = $1, accepted_at = NOW() WHERE id = $2',
      ['accepted', invitation.id]
    );
    
    // Create welcome notification for inviter
    await createTeamNotification(
      invitation.invited_by,
      'team_member_joined',
      'New Team Member Joined',
      `${userData.first_name} ${userData.last_name} has joined your team`,
      null,
      newUser.id
    );
    
    // Log activity
    await logActivity(null, newUser.id, 'team_member_joined', `${userData.first_name} ${userData.last_name} joined the team`);
    
    return newUser;
  },
  
  // Assign team member to deal
  assignUserToDeal: async (targetCompanyId, userId, assignedBy, assignmentData) => {
    // Verify assigner has permission
    const assigner = await getUserWithPermissions(assignedBy);
    if (!assigner.can_manage_users && !assigner.is_company_admin) {
      // Check if assigner is assigned to this deal as lead
      const dealAssignment = await db.query(
        'SELECT assignment_role FROM user_deal_assignments WHERE user_id = $1 AND target_company_id = $2 AND is_active = true',
        [assignedBy, targetCompanyId]
      );
      
      if (dealAssignment.rows.length === 0 || dealAssignment.rows[0].assignment_role !== 'lead') {
        throw new Error('Insufficient permissions to assign users to this deal');
      }
    }
    
    // Verify target user exists and is in same company
    const targetUser = await db.query(
      'SELECT * FROM users WHERE id = $1 AND user_company_id = $2 AND is_active = true',
      [userId, assigner.user_company_id]
    );
    
    if (targetUser.rows.length === 0) {
      throw new Error('User not found or not in your company');
    }
    
    // Create or update assignment
    const assignmentQuery = `
      INSERT INTO user_deal_assignments (
        user_id, target_company_id, assigned_by, assignment_role,
        can_edit_company, can_upload_documents, can_create_activities, can_manage_contacts
      ) VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
      ON CONFLICT (user_id, target_company_id) 
      DO UPDATE SET 
        assignment_role = EXCLUDED.assignment_role,
        assigned_by = EXCLUDED.assigned_by,
        assignment_date = NOW(),
        is_active = true,
        can_edit_company = EXCLUDED.can_edit_company,
        can_upload_documents = EXCLUDED.can_upload_documents,
        can_create_activities = EXCLUDED.can_create_activities,
        can_manage_contacts = EXCLUDED.can_manage_contacts
      RETURNING *
    `;
    
    const assignmentValues = [
      userId,
      targetCompanyId,
      assignedBy,
      assignmentData.role || 'analyst',
      assignmentData.can_edit_company !== false,
      assignmentData.can_upload_documents !== false,
      assignmentData.can_create_activities !== false,
      assignmentData.can_manage_contacts !== false
    ];
    
    const result = await db.query(assignmentQuery, assignmentValues);
    
    // Get company name for notification
    const company = await db.query('SELECT company_name FROM target_companies WHERE id = $1', [targetCompanyId]);
    
    // Create notification for assigned user
    await createTeamNotification(
      userId,
      'deal_assigned',
      'Assigned to Deal',
      `You have been assigned to ${company.rows[0].company_name}`,
      targetCompanyId,
      assignedBy
    );
    
    // Log activity
    await logActivity(targetCompanyId, assignedBy, 'team_member_assigned', `Assigned ${targetUser.rows[0].first_name} ${targetUser.rows[0].last_name} to deal`);
    
    return result.rows[0];
  },
  
  // Get team members with their roles and permissions
  getTeamMembers: async (userCompanyId, requestingUserId) => {
    const teamQuery = `
      SELECT 
        u.id,
        u.email,
        u.first_name,
        u.last_name,
        u.user_role,
        u.is_company_admin,
        u.is_active,
        u.last_login,
        u.created_at,
        ur.role_description,
        ur.can_manage_users,
        ur.can_create_deals,
        ur.can_view_financials,
        COUNT(uda.target_company_id) as assigned_deals_count
      FROM users u
      LEFT JOIN user_roles ur ON u.user_role = ur.role_name
      LEFT JOIN user_deal_assignments uda ON u.id = uda.user_id AND uda.is_active = true
      WHERE u.user_company_id = $1 AND u.is_active = true
      GROUP BY u.id, ur.role_description, ur.can_manage_users, ur.can_create_deals, ur.can_view_financials
      ORDER BY u.is_company_admin DESC, u.created_at ASC
    `;
    
    const result = await db.query(teamQuery, [userCompanyId]);
    return result.rows;
  }
};

const notificationManager = {
  // Create team notification
  createTeamNotification: async (targetUserId, notificationType, title, message, targetCompanyId = null, fromUserId = null) => {
    const notificationQuery = `
      INSERT INTO team_notifications (
        user_company_id, target_user_id, from_user_id, notification_type,
        notification_title, notification_message, target_company_id
      ) VALUES ($1, $2, $3, $4, $5, $6, $7)
      RETURNING *
    `;
    
    // Get user's company ID
    const userCompany = await db.query('SELECT user_company_id FROM users WHERE id = $1', [targetUserId]);
    
    const values = [
      userCompany.rows[0].user_company_id,
      targetUserId,
      fromUserId,
      notificationType,
      title,
      message,
      targetCompanyId
    ];
    
    const result = await db.query(notificationQuery, values);
    
    // Send email notification if user preferences allow
    await sendEmailNotification(result.rows[0]);
    
    return result.rows[0];
  },
  
  // Get user notifications
  getUserNotifications: async (userId, limit = 50) => {
    const notificationsQuery = `
      SELECT 
        tn.*,
        fu.first_name as from_first_name,
        fu.last_name as from_last_name,
        tc.company_name
      FROM team_notifications tn
      LEFT JOIN users fu ON tn.from_user_id = fu.id
      LEFT JOIN target_companies tc ON tn.target_company_id = tc.id
      WHERE tn.target_user_id = $1
      ORDER BY tn.created_at DESC
      LIMIT $2
    `;
    
    const result = await db.query(notificationsQuery, [userId, limit]);
    return result.rows;
  },
  
  // Mark notification as read
  markNotificationRead: async (notificationId, userId) => {
    const result = await db.query(
      'UPDATE team_notifications SET is_read = true, read_at = NOW() WHERE id = $1 AND target_user_id = $2 RETURNING *',
      [notificationId, userId]
    );
    
    return result.rows[0];
  }
};

const getUserWithPermissions = async (userId) => {
  const userQuery = `
    SELECT 
      u.*,
      ur.can_manage_users,
      ur.can_manage_company,
      ur.can_create_deals,
      ur.can_edit_all_deals,
      ur.can_delete_deals,
      ur.can_upload_documents,
      ur.can_create_offers,
      ur.can_view_financials,
      ur.can_access_analytics,
      ur.can_export_data
    FROM users u
    LEFT JOIN user_roles ur ON u.user_role = ur.role_name
    WHERE u.id = $1 AND u.is_active = true
  `;
  
  const result = await db.query(userQuery, [userId]);
  return result.rows[0];
};

// Team member invitation
document.getElementById('invite-member-form').addEventListener('submit', async (e) => {
  e.preventDefault();
  
  const formData = new FormData(e.target);
  const invitationData = Object.fromEntries(formData.entries());
  
  try {
    const response = await fetch('/api/team/invite', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(invitationData)
    });
    
    const result = await response.json();
    
    if (response.ok) {
      showNotification(`Invitation sent to ${invitationData.email}`, 'success');
      closeModal('invite-member-modal');
      refreshTeamMembers();
    } else {
      showNotification(result.error || 'Failed to send invitation', 'error');
    }
    
  } catch (error) {
    console.error('Invitation failed:', error);
    showNotification('Failed to send invitation', 'error');
  }
});

const updateRolePermissionsDisplay = (role) => {
  const permissionsContainer = document.getElementById('role-permissions');
  
  const rolePermissions = {
    'analyst': ['Upload documents', 'Create activities', 'View assigned deals'],
    'senior_analyst': ['Create deals', 'Upload documents', 'Create activities', 'View financials'],
    'deal_manager': ['Manage team assignments', 'Create/edit deals', 'View all company data', 'Create offers'],
    'company_admin': ['Full system access', 'User management', 'Billing access', 'All permissions']
  };
  
  const permissions = rolePermissions[role] || [];
  
  permissionsContainer.innerHTML = `
    <div class="permissions-preview">
      <h5>Role Permissions:</h5>
      <ul>
        ${permissions.map(permission => `<li>✓ ${permission}</li>`).join('')}
      </ul>
    </div>
  `;
};

// Deal assignment management
const showDealAssignmentModal = (dealId, dealName, dealStage) => {
  document.getElementById('assignment-deal-name').textContent = dealName;
  document.getElementById('assignment-deal-stage').textContent = `Stage: ${dealStage}`;
  
  // Load current assignments
  loadCurrentAssignments(dealId);

const loadCurrentAssignments = async (dealId) => {
  try {
    const response = await fetch(`/api/deals/${dealId}/team`);
    const assignments = await response.json();
    
    if (response.ok) {
      // Update assignment dropdowns based on current assignments
      assignments.forEach(assignment => {
        const roleSelect = document.querySelector(`[data-user-id="${assignment.user_id}"]`);
        if (roleSelect) {
          roleSelect.value = assignment.assignment_role;
          
          // Show permissions checkboxes
          const permissionsDiv = roleSelect.parentElement.querySelector('.assignment-permissions');
          permissionsDiv.style.display = 'block';
          
          // Set permission checkboxes
          permissionsDiv.querySelector('[name="can_edit_company"]').checked = assignment.can_edit_company;
          permissionsDiv.querySelector('[name="can_upload_documents"]').checked = assignment.can_upload_documents;
          permissionsDiv.querySelector('[name="can_create_activities"]').checked = assignment.can_create_activities;
          permissionsDiv.querySelector('[name="can_manage_contacts"]').checked = assignment.can_manage_contacts;
        }
      });
    }
    
  } catch (error) {
    console.error('Failed to load assignments:', error);
  }
};

const saveTeamAssignments = async () => {
  const dealId = getCurrentDealId(); // Get from modal context
  const assignments = [];
  
  // Collect all assignment data
  document.querySelectorAll('.assignment-role').forEach(select => {
    const userId = select.dataset.userId;
    const role = select.value;
    
    if (role) {
      const permissionsDiv = select.parentElement.querySelector('.assignment-permissions');
      const permissions = {
        can_edit_company: permissionsDiv.querySelector('[name="can_edit_company"]').checked,
        can_upload_documents: permissionsDiv.querySelector('[name="can_upload_documents"]').checked,
        can_create_activities: permissionsDiv.querySelector('[name="can_create_activities"]').checked,
        can_manage_contacts: permissionsDiv.querySelector('[name="can_manage_contacts"]').checked
      };
      
      assignments.push({
        user_id: parseInt(userId),
        role: role,
        ...permissions
      });
    }
  });
  
  try {
    const response = await fetch(`/api/deals/${dealId}/assign`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ assignments })
    });
    
    if (response.ok) {
      showNotification('Team assignments saved successfully', 'success');
      closeModal('assign-deal-modal');
      refreshDealTeam(dealId);
    } else {
      showNotification('Failed to save team assignments', 'error');
    }
    
  } catch (error) {
    console.error('Assignment save failed:', error);
    showNotification('Failed to save team assignments', 'error');
  }
};

const markNotificationRead = async (notificationId) => {
  try {
    const response = await fetch(`/api/notifications/${notificationId}/read`, {
      method: 'PUT'
    });
    
    if (response.ok) {
      // Update UI to show as read
      const notificationElement = document.querySelector(`[data-notification-id="${notificationId}"]`);
      if (notificationElement) {
        notificationElement.classList.remove('unread');
        notificationElement.querySelector('.notification-actions').innerHTML = '';
      }
    }
    
  } catch (error) {
    console.error('Failed to mark notification as read:', error);
  }
};
