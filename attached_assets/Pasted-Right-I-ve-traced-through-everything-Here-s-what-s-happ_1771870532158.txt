Problem 3: The source model should be auto-locked when duplicating
In server/services/deal-model.service.ts, the duplicateModel method creates a copy but doesn't lock the source model. The user's previous version should be automatically locked to prevent accidental changes.
Fix: In the duplicateModel method, after creating the duplicate, lock the source model:
jsasync duplicateModel(
  id: number,
  userCompanyId: number,
  newName?: string
): Promise<DealModelWithDetails & { calculations: CalculationResult }> {
  const model = await dealModelRepository.getById(id);
  if (!model) throw new NotFoundError('Deal model not found');
  await verifyOwnership(model.targetCompanyId, userCompanyId);

  // Determine next version number
  const allModels = await dealModelRepository.listByCompany(model.targetCompanyId);
  const maxVersion = Math.max(...allModels.map(m => m.version ?? 1), 0);
  const nextVersion = maxVersion + 1;

  const name = newName || `Deal Model v${nextVersion}`;
  const newModel = await dealModelRepository.duplicate(id, name, model.createdBy);

  // Auto-lock the source model so previous version is preserved
  if (!model.isLocked) {
    await dealModelRepository.lock(id);
    // Invalidation will happen client-side via queryClient
  }

  const fullModel = await dealModelRepository.getByIdWithDetails(newModel.id);
  if (!fullModel) throw new NotFoundError('Duplicated deal model not found');

  const calculations = await this.recalculate(fullModel);

  return { ...fullModel, calculations };
}
