// Pipeline overview widget
const getPipelineOverviewWidget = async (userCompanyId) => {
  const pipelineData = await db.query(`
    SELECT 
      stage,
      COUNT(*) as count,
      AVG(EXTRACT(days FROM (NOW() - stage_updated_at))) as avg_days
    FROM target_companies 
    WHERE user_company_id = $1 
    AND stage NOT IN ('Completion', 'Not Qualified')
    GROUP BY stage
    ORDER BY 
      CASE stage
        WHEN 'New Target' THEN 1
        WHEN 'Initial Contact' THEN 2
        WHEN 'Documents Requested' THEN 3
        WHEN 'Under Review' THEN 4
        WHEN 'Offer Stage' THEN 5
        WHEN 'Negotiation' THEN 6
        WHEN 'Due Diligence' THEN 7
        WHEN 'Legal & Accounting' THEN 8
      END
  `, [userCompanyId]);
  
  const totalActive = pipelineData.rows.reduce((sum, stage) => sum + parseInt(stage.count), 0);
  
  return {
    stages: pipelineData.rows,
    totalActive,
    lastUpdated: new Date()
  };
};

// Recent activities widget
const getRecentActivitiesWidget = async (userCompanyId) => {
  const activities = await db.query(`
    SELECT 
      a.*,
      tc.company_name,
      u.first_name || ' ' || u.last_name as user_name,
      cont.first_name || ' ' || cont.last_name as contact_name
    FROM activities a
    JOIN target_companies tc ON a.target_company_id = tc.id
    JOIN users u ON a.user_id = u.id
    LEFT JOIN target_contacts cont ON a.contact_id = cont.id
    WHERE tc.user_company_id = $1
    ORDER BY a.created_at DESC
    LIMIT 10
  `, [userCompanyId]);
  
  return {
    activities: activities.rows,
    lastUpdated: new Date()
  };
};

// Hot deals widget (high priority or time-sensitive)
const getHotDealsWidget = async (userCompanyId) => {
  const hotDeals = await db.query(`
    SELECT 
      tc.*,
      EXTRACT(days FROM (NOW() - tc.stage_updated_at)) as days_in_stage,
      COUNT(a.id) as recent_activity_count,
      MAX(a.activity_date) as last_activity_date
    FROM target_companies tc
    LEFT JOIN activities a ON tc.id = a.target_company_id 
      AND a.activity_date >= CURRENT_DATE - INTERVAL '7 days'
    WHERE tc.user_company_id = $1
    AND tc.stage NOT IN ('Completion', 'Not Qualified')
    GROUP BY tc.id
    HAVING 
      EXTRACT(days FROM (NOW() - tc.stage_updated_at)) > 14 -- Stuck in stage
      OR COUNT(a.id) = 0 -- No recent activity
      OR tc.stage IN ('Offer Stage', 'Negotiation', 'Due Diligence') -- Critical stages
    ORDER BY 
      CASE 
        WHEN tc.stage IN ('Offer Stage', 'Negotiation') THEN 1
        WHEN EXTRACT(days FROM (NOW() - tc.stage_updated_at)) > 30 THEN 2
        ELSE 3
      END,
      tc.stage_updated_at ASC
    LIMIT 5
  `, [userCompanyId]);
  
  return {
    deals: hotDeals.rows,
    lastUpdated: new Date()
  };
};

Real-time Updates Implementation
// Server-sent events for real-time updates
const setupRealTimeUpdates = () => {
  const eventSource = new EventSource('/api/dashboard/live-updates');
  
  eventSource.onmessage = (event) => {
    const update = JSON.parse(event.data);
    handleRealTimeUpdate(update);
  };
  
  eventSource.onerror = (error) => {
    console.error('Real-time connection error:', error);
    // Attempt to reconnect after 5 seconds
    setTimeout(setupRealTimeUpdates, 5000);
  };
};

const handleRealTimeUpdate = (update) => {
  switch (update.type) {
    case 'stage_change':
      updatePipelineWidget();
      showNotification(`${update.company_name} moved to ${update.new_stage}`, 'info');
      break;
      
    case 'new_activity':
      updateActivitiesWidget();
      updateHotDealsWidget();
      break;
      
    case 'new_deal':
      updatePipelineWidget();
      updateStatsWidget();
      break;
      
    case 'reminder_due':
      updateRemindersWidget();
      showNotification(`Reminder: ${update.title}`, 'warning');
      break;
  }
};

// Widget refresh functions
const refreshWidget = async (widgetType) => {
  try {
    showWidgetLoading(widgetType);
    
    const response = await fetch(`/api/dashboard/data/${widgetType}`);
    const data = await response.json();
    
    updateWidgetContent(widgetType, data);
    hideWidgetLoading(widgetType);
    
  } catch (error) {
    console.error(`Failed to refresh ${widgetType} widget:`, error);
    showWidgetError(widgetType);
  }
};

const showCustomizationModal = () => {
  const modal = document.getElementById('customization-modal');
  modal.style.display = 'block';
  
  // Load current widget configuration
  loadWidgetConfiguration();
};

const saveWidgetConfiguration = async (config) => {
  try {
    await fetch('/api/dashboard/widgets', {
      method: 'PUT',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(config)
    });
    
    // Reload dashboard with new configuration
    location.reload();
    
  } catch (error) {
    console.error('Failed to save widget configuration:', error);
    showNotification('Failed to save dashboard configuration', 'error');
  }
};
