     import Stripe from 'stripe';
     import { eq, inArray } from 'drizzle-orm';
     import { db } from '../db';
     import { subscriptions } from '@shared/schema/billing';
     import { config } from '../config';
     import { billingRepository } from '../repositories/billing.repository';
     
     // One-shot reconciliation script for subscriptions affected by the
     // pre-#122 invoice.paid bug that wrote current_period_start =
     // current_period_end. Fetches canonical period dates from Stripe
     // and writes them back. Idempotent — skips rows where
     // current_period_end > current_period_start (already fixed) and
     // skips rows where Stripe returns the same collided dates we
     // already have.
     
     const TARGET_SUBSCRIPTION_IDS = [38, 40, 44];
     
     export async function reconcileSubscriptionPeriods(): Promise<{
       reconciled: number;
       skipped_already_fixed: number;
       skipped_stripe_no_change: number;
       errors: number;
     }> {
       console.log('[RECONCILE] Starting subscription period reconciliation...');
       
       if (!config.stripe.secretKey) {
         console.error('[RECONCILE] STRIPE_SECRET_KEY not configured');
         throw new Error('STRIPE_SECRET_KEY required');
       }
       const stripe = new Stripe(config.stripe.secretKey);
       
       const rows = await db
         .select()
         .from(subscriptions)
         .where(inArray(subscriptions.id, TARGET_SUBSCRIPTION_IDS));
       
       console.log(`[RECONCILE] Found ${rows.length} target rows on prod (expected 3)`);
       
       let reconciled = 0;
       let skipped_already_fixed = 0;
       let skipped_stripe_no_change = 0;
       let errors = 0;
       
       for (const row of rows) {
         const tag = `sub ${row.id} (company ${row.userCompanyId})`;
         
         try {
           // Idempotency check 1: already fixed?
           if (
             row.currentPeriodStart &&
             row.currentPeriodEnd &&
             row.currentPeriodEnd.getTime() > row.currentPeriodStart.getTime()
           ) {
             console.log(`[RECONCILE] ${tag}: SKIP — already has valid period (start=${row.currentPeriodStart.toISOString()}, end=${row.currentPeriodEnd.toISOString()})`);
             skipped_already_fixed++;
             continue;
           }
           
           if (!row.stripeSubscriptionId) {
             console.log(`[RECONCILE] ${tag}: SKIP — no stripe_subscription_id`);
             errors++;
             continue;
           }
           
           console.log(`[RECONCILE] ${tag}: fetching ${row.stripeSubscriptionId} from Stripe...`);
           const stripeSub = await stripe.subscriptions.retrieve(row.stripeSubscriptionId);
           const primaryItem = stripeSub.items.data[0];
           
           if (!primaryItem) {
             console.error(`[RECONCILE] ${tag}: Stripe sub has no items, cannot read period`);
             errors++;
             continue;
           }
           
           const stripePeriodStart = primaryItem.current_period_start
             ? new Date(primaryItem.current_period_start * 1000)
             : null;
           const stripePeriodEnd = primaryItem.current_period_end
             ? new Date(primaryItem.current_period_end * 1000)
             : null;
           
           if (!stripePeriodStart || !stripePeriodEnd) {
             console.error(`[RECONCILE] ${tag}: Stripe item missing current_period_start/end`);
             errors++;
             continue;
           }
           
           // Idempotency check 2: did Stripe also have the collided dates?
           // (This shouldn't happen but is a defensive guard.)
           if (stripePeriodEnd.getTime() <= stripePeriodStart.getTime()) {
             console.error(`[RECONCILE] ${tag}: Stripe also reports collided dates (start=${stripePeriodStart.toISOString()}, end=${stripePeriodEnd.toISOString()}), refusing to write`);
             skipped_stripe_no_change++;
             continue;
           }
           
           console.log(`[RECONCILE] ${tag}: DB had start=${row.currentPeriodStart?.toISOString()}, end=${row.currentPeriodEnd?.toISOString()}`);
           console.log(`[RECONCILE] ${tag}: Stripe says start=${stripePeriodStart.toISOString()}, end=${stripePeriodEnd.toISOString()}`);
           
           await db
             .update(subscriptions)
             .set({
               currentPeriodStart: stripePeriodStart,
               currentPeriodEnd: stripePeriodEnd,
             })
             .where(eq(subscriptions.id, row.id));
           
           // Audit row in subscription_events
           await billingRepository.logEvent({
             userCompanyId: row.userCompanyId,
             stripeEventId: null,
             eventType: 'admin.period_reconciliation',
             previousStatus: row.status,
             previousPlanKey: row.planKey,
             newStatus: row.status,
             newPlanKey: row.planKey,
             payload: {
               source: 'reconcile-subscription-periods script',
               previousCurrentPeriodStart: row.currentPeriodStart?.toISOString() ?? null,
               previousCurrentPeriodEnd: row.currentPeriodEnd?.toISOString() ?? null,
               newCurrentPeriodStart: stripePeriodStart.toISOString(),
               newCurrentPeriodEnd: stripePeriodEnd.toISOString(),
               stripeSubscriptionId: row.stripeSubscriptionId,
             },
           });
           
           console.log(`[RECONCILE] ${tag}: WRITTEN — period now ${stripePeriodStart.toISOString()} to ${stripePeriodEnd.toISOString()}`);
           reconciled++;
         } catch (err) {
           console.error(`[RECONCILE] ${tag}: ERROR`, err);
           errors++;
         }
       }
       
       console.log(`[RECONCILE] Done — reconciled: ${reconciled}, skipped_already_fixed: ${skipped_already_fixed}, skipped_stripe_no_change: ${skipped_stripe_no_change}, errors: ${errors}`);
       return { reconciled, skipped_already_fixed, skipped_stripe_no_change, errors };
     }
     
     if (process.argv[1]?.includes('reconcile-subscription-periods')) {
       reconcileSubscriptionPeriods()
         .then((result) => {
           console.log('[RECONCILE] Final result:', result);
           process.exit(0);
         })
         .catch((err) => {
           console.error('[RECONCILE] Failed:', err);
           process.exit(1);
         });
     }

DECISIONS NEEDED FROM ADAM:

D127.1. Where is the script run from?
  (A) Replit prod shell directly: 
      `npx tsx server/scripts/reconcile-subscription-periods.ts`
      (Simpler. Same pattern as backfill-company-accounts. Connects
      to whichever DATABASE_URL is set in the Replit env.)
  (B) From local with prod DATABASE_URL temporarily exported.
      (More steps, slightly safer because a typo in the file affects
      local first.)
  Recommended: (A). Convention. Lower friction. tsc has already
  verified the file compiles before run.

D127.2. Should the script use a transaction per row?
  (A) Two separate writes per row (UPDATE subscriptions, then INSERT
      subscription_events). Current proposal. If billingRepository.
      logEvent fails after the UPDATE succeeded, the audit row is
      missing but the data fix is correct.
  (B) Wrap both writes in a db.transaction() per row.
  Recommended: (A). billingRepository.logEvent failures are extremely
  rare (just a simple INSERT). The data fix landing without the audit
  row is the lesser evil compared to half-applied transactions.

D127.3. Should we run a dry-run mode first?
  (A) No dry-run. Three rows, well-bounded scope, idempotent on re-
      run, clear log output. Just run it.
  (B) Add a DRY_RUN env var that does everything except the writes.
  Recommended: (A). The idempotency guards protect against accidents,
  and adding dry-run for 3 rows is overkill. The output of running
  it once tells us everything we need.
