4. File Upload Implementation
const multer = require('multer');
const path = require('path');
const fs = require('fs').promises;

// Configure multer for file uploads
const storage = multer.diskStorage({
  destination: async (req, file, cb) => {
    const companyId = req.params.id;
    const uploadPath = path.join(__dirname, '../uploads/companies', companyId);
    
    // Create directory if it doesn't exist
    try {
      await fs.mkdir(uploadPath, { recursive: true });
      cb(null, uploadPath);
    } catch (error) {
      cb(error);
    }
  },
  filename: (req, file, cb) => {
    // Generate unique filename
    const uniqueSuffix = Date.now() + '-' + Math.round(Math.random() * 1E9);
    const extension = path.extname(file.originalname);
    cb(null, `doc-${uniqueSuffix}${extension}`);
  }
});

const fileFilter = (req, file, cb) => {
  const allowedTypes = [
    'application/pdf',
    'application/msword',
    'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
    'application/vnd.ms-excel',
    'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
    'application/vnd.ms-powerpoint',
    'application/vnd.openxmlformats-officedocument.presentationml.presentation',
    'text/plain',
    'image/jpeg',
    'image/png'
  ];
  
  if (allowedTypes.includes(file.mimetype)) {
    cb(null, true);
  } else {
    cb(new Error('File type not supported'), false);
  }
};

const upload = multer({
  storage: storage,
  fileFilter: fileFilter,
  limits: {
    fileSize: 10 * 1024 * 1024 // 10MB limit
  }
});

// Upload document endpoint
const uploadDocument = async (req, res) => {
  try {
    const targetCompanyId = req.params.id;
    const userId = req.user.id;
    
    // Verify user has access to this company
    const company = await getTargetCompanyById(targetCompanyId, userId);
    if (!company) {
      return res.status(404).json({ error: 'Company not found or access denied' });
    }
    
    // Handle file upload
    upload.single('document')(req, res, async (err) => {
      if (err) {
        if (err.code === 'LIMIT_FILE_SIZE') {
          return res.status(400).json({ error: 'File size too large. Maximum 10MB allowed.' });
        }
        return res.status(400).json({ error: err.message });
      }
      
      if (!req.file) {
        return res.status(400).json({ error: 'No file uploaded' });
      }
      
      // Save document record to database
      const documentData = {
        target_company_id: targetCompanyId,
        uploaded_by: userId,
        document_name: req.body.document_name || req.file.originalname,
        original_filename: req.file.originalname,
        file_path: req.file.path,
        file_size: req.file.size,
        file_type: req.file.mimetype,
        document_category: req.body.document_category || 'Other',
        description: req.body.description || '',
        is_confidential: req.body.is_confidential !== 'false'
      };
      
      const result = await createDocument(documentData);
      
      res.json({
        success: true,
        document: result,
        message: 'Document uploaded successfully'
      });
    });
    
  } catch (error) {
    console.error('Document upload failed:', error);
    res.status(500).json({ error: 'Upload failed' });
  }
};

5. Document Listing and Management
const getCompanyDocuments = async (targetCompanyId, userId) => {
  // Verify access
  const company = await getTargetCompanyById(targetCompanyId, userId);
  if (!company) {
    throw new Error('Company not found or access denied');
  }
  
  const query = `
    SELECT 
      d.*,
      u.first_name || ' ' || u.last_name as uploaded_by_name,
      dc.description as category_description
    FROM documents d
    JOIN users u ON d.uploaded_by = u.id
    LEFT JOIN document_categories dc ON d.document_category = dc.category_name
    WHERE d.target_company_id = $1
    ORDER BY d.document_category, d.upload_date DESC
  `;
  
  const result = await db.query(query, [targetCompanyId]);
  
  // Group documents by category
  const groupedDocuments = {};
  result.rows.forEach(doc => {
    if (!groupedDocuments[doc.document_category]) {
      groupedDocuments[doc.document_category] = [];
    }
    groupedDocuments[doc.document_category].push(doc);
  });
  
  return groupedDocuments;
};

const downloadDocument = async (documentId, userId) => {
  // Verify user has access to this document
  const query = `
    SELECT d.*, tc.user_company_id
    FROM documents d
    JOIN target_companies tc ON d.target_company_id = tc.id
    JOIN users u ON tc.user_company_id = u.user_company_id
    WHERE d.id = $1 AND u.id = $2
  `;
  
  const result = await db.query(query, [documentId, userId]);
  
  if (result.rows.length === 0) {
    throw new Error('Document not found or access denied');
  }
  
  const document = result.rows[0];
  
  // Log document access
  await logDocumentAccess(documentId, userId, 'download');
  
  // Update access count
  await db.query(
    'UPDATE documents SET access_count = access_count + 1, last_accessed = NOW() WHERE id = $1',
    [documentId]
  );
  
  return document;
};

const logDocumentAccess = async (documentId, userId, accessType, ipAddress = null) => {
  await db.query(
    'INSERT INTO document_access_log (document_id, accessed_by, access_type, ip_address) VALUES ($1, $2, $3, $4)',
    [documentId, userId, accessType, ipAddress]
  );
};

// Handle file upload form
document.getElementById('document-upload-form').addEventListener('submit', async (e) => {
  e.preventDefault();
  
  const formData = new FormData(e.target);
  const companyId = getCurrentCompanyId();
  
  // Show upload progress
  showUploadProgress();
  
  try {
    const response = await fetch(`/api/target-companies/${companyId}/documents`, {
      method: 'POST',
      body: formData
    });
    
    const result = await response.json();
    
    if (result.success) {
      showNotification('Document uploaded successfully', 'success');
      hideUploadForm();
      refreshDocumentsList();
      
      // Reset form
      e.target.reset();
    } else {
      showNotification(result.error || 'Upload failed', 'error');
    }
    
  } catch (error) {
    console.error('Upload failed:', error);
    showNotification('Upload failed. Please try again.', 'error');
  } finally {
    hideUploadProgress();
  }
});

// File input change handler
document.getElementById('document-file').addEventListener('change', (e) => {
  const file = e.target.files[0];
  if (file) {
    // Auto-populate document name if not set
    const nameInput = document.getElementById('document-name');
    if (!nameInput.value) {
      nameInput.value = file.name.replace(/\.[^/.]+$/, ''); // Remove extension
    }
    
    // Validate file size
    if (file.size > 10 * 1024 * 1024) {
      showNotification('File size too large. Maximum 10MB allowed.', 'error');
      e.target.value = '';
      return;
    }
    
    // Show file info
    updateFileInfo(file);
  }
});

const updateFileInfo = (file) => {
  const fileInfo = document.querySelector('.file-info');
  const sizeInMB = (file.size / (1024 * 1024)).toFixed(2);
  fileInfo.innerHTML = `
    <small>
      Selected: ${file.name} (${sizeInMB} MB)<br>
      Maximum file size: 10MB. Supported formats: PDF, DOC, XLS, PPT, TXT, JPG, PNG
    </small>
  `;
};

// Document actions
const downloadDocument = async (documentId) => {
  try {
    const response = await fetch(`/api/documents/${documentId}/download`);
    
    if (response.ok) {
      const blob = await response.blob();
      const url = window.URL.createObjectURL(blob);
      
      // Get filename from response headers
      const contentDisposition = response.headers.get('Content-Disposition');
      const filename = contentDisposition 
        ? contentDisposition.split('filename=')[1].replace(/"/g, '')
        : 'document';
      
      // Create download link
      const a = document.createElement('a');
      a.href = url;
      a.download = filename;
      document.body.appendChild(a);
      a.click();
      document.body.removeChild(a);
      window.URL.revokeObjectURL(url);
      
    } else {
      showNotification('Download failed', 'error');
    }
    
  } catch (error) {
    console.error('Download failed:', error);
    showNotification('Download failed', 'error');
  }
};

const deleteDocument = async (documentId) => {
  if (!confirm('Are you sure you want to delete this document? This action cannot be undone.')) {
    return;
  }
  
  try {
    const response = await fetch(`/api/documents/${documentId}`, {
      method: 'DELETE'
    });
    
    if (response.ok) {
      showNotification('Document deleted successfully', 'success');
      refreshDocumentsList();
    } else {
      showNotification('Failed to delete document', 'error');
    }
    
  } catch (error) {
    console.error('Delete failed:', error);
    showNotification('Delete failed', 'error');
  }
};
