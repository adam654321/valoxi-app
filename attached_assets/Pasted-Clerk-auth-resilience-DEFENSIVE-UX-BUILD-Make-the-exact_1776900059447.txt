Current file is 27 lines. Replace entirely with:

import { useMutation } from "@tanstack/react-query";
import { useClerk } from "@clerk/react";
import { queryClient, apiRequest } from "@/lib/queryClient";
import { Button } from "@/components/ui/button";
import { useToast } from "@/hooks/use-toast";

export default function AccountSetupPending() {
  const { signOut } = useClerk();
  const { toast } = useToast();

  const clerkEnabled = import.meta.env.VITE_CLERK_ENABLED === 'true';

  const signOutMutation = useMutation({
    mutationFn: async () => {
      try {
        await apiRequest("POST", "/api/v2/auth/logout");
      } catch (err) {
        console.warn('[account-setup-pending] Server logout failed, continuing with Clerk signOut', err);
      }
      if (clerkEnabled) {
        await signOut();
      }
      window.location.href = "/login";
    },
    onError: (err: any) => {
      toast({
        title: "Sign-out failed",
        description: err?.message || "Please try again or clear your browser cookies.",
        variant: "destructive",
      });
    },
  });

  return (
    <div className="min-h-screen flex items-center justify-center py-12 px-4 sm:px-6 lg:px-8 bg-background">
      <div className="max-w-md w-full rounded-lg border bg-card p-8 text-center">
        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-primary mx-auto mb-4" />
        <h1 className="text-xl font-semibold mb-3">Setting up your account</h1>
        <p className="text-sm text-muted-foreground mb-6">
          Your account is being linked. This usually takes a few seconds. The page will refresh
          automatically once setup is complete.
        </p>
        <div className="flex flex-col gap-2">
          <Button
            onClick={() =>
              queryClient.invalidateQueries({ queryKey: ["/api/v2/auth/me"] })
            }
          >
            Retry now
          </Button>
          <Button
            variant="outline"
            onClick={() => signOutMutation.mutate()}
            disabled={signOutMutation.isPending}
          >
            {signOutMutation.isPending ? "Signing out..." : "Sign out and start over"}
          </Button>
        </div>
        <p className="mt-4 text-sm text-muted-foreground">
          Still stuck?{" "}
          <a href="mailto:support@valoxi.ai" className="underline">
            Contact support
          </a>
        </p>
      </div>
    </div>
  );
}

Note: this escape calls BOTH server /logout AND clerk signOut(), then hard-navigates to /login (the non-Clerk fallback). That path is known to work regardless of Clerk state.

CHANGE 3 of 4: sign-in.tsx — pre-flight clear stuck session
===========================================================================

File: client/src/pages/auth/sign-in.tsx

Replace the current 19-line file with:

import { useEffect, useState } from "react";
import { SignIn } from "@clerk/react";
import { useClerk, useAuth as useClerkAuth } from "@clerk/react";
import { ValoxiLogo } from "@/components/valoxi-logo";
import { valoxiClerkAppearance } from "@/lib/clerk-appearance";
import { Button } from "@/components/ui/button";

export default function SignInPage() {
  const { signOut } = useClerk();
  const { isSignedIn, isLoaded } = useClerkAuth();
  const [preflightComplete, setPreflightComplete] = useState(false);

  // Pre-flight: if a Clerk session exists when the user lands on /sign-in,
  // that is a stuck state (they should not see /sign-in if they are signed in;
  // they should be on /). Clear the session before rendering the Clerk component.
  useEffect(() => {
    if (!isLoaded) return;

    const checkStuckSession = async () => {
      if (isSignedIn) {
        try {
          // Verify with our backend whether the Clerk session resolves to a linked user.
          const res = await fetch("/api/v2/auth/me", { credentials: "include" });
          if (!res.ok) {
            // Clerk says signed in but backend does not agree. Clear the stuck state.
            console.warn('[sign-in] Stuck Clerk session detected (Clerk signed in, backend not linked). Clearing.');
            await signOut();
          }
        } catch (err) {
          console.warn('[sign-in] Backend check failed, clearing Clerk session defensively', err);
          await signOut();
        }
      }
      setPreflightComplete(true);
    };

    checkStuckSession();
  }, [isLoaded, isSignedIn, signOut]);

  const handleStartOver = async () => {
    try {
      await signOut();
    } catch (err) {
      console.error('[sign-in] signOut failed', err);
    }
    window.location.reload();
  };

  if (!isLoaded || !preflightComplete) {
    return (
      <div className="min-h-screen flex items-center justify-center">
        <div className="animate-spin rounded-full h-6 w-6 border-b-2 border-primary" />
      </div>
    );
  }

  return (
    <div className="min-h-screen flex items-center justify-center py-12 px-4 sm:px-6 lg:px-8 bg-background">
      <div className="flex flex-col items-center gap-4">
        <ValoxiLogo />
        <SignIn
          appearance={valoxiClerkAppearance}
          signUpUrl="/sign-up"
          forceRedirectUrl="/"
          fallbackRedirectUrl="/"
        />
        <div className="flex flex-col items-center gap-2 mt-2 text-sm text-muted-foreground">
          <button
            type="button"
            onClick={handleStartOver}
            className="underline hover:text-foreground"
          >
            Having trouble? Start over
          </button>
          <span>
            or use the{" "}
            <a href="/login" className="underline hover:text-foreground">
              classic login
            </a>
          </span>
        </div>
      </div>
    </div>
  );
}

Note on the "classic login" link: /login still works as a Clerk-free fallback (Express session auth). This gives any stuck user an immediate escape hatch. It will stop working if/when /login is retired, but that is not in scope for this cycle.
