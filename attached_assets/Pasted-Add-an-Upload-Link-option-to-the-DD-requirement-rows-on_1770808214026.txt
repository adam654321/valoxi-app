In the main DueDiligenceTab component, add:
```typescript
const handleUploadAndLink = async (requirementId: number, file: File, docName: string) => {
  // Step 1: Upload the document
  const formData = new FormData();
  formData.append('file', file);  // CHECK: match the exact field name from DocumentManagement
  formData.append('documentName', docName);
  formData.append('documentCategory', 'Due Diligence');

  const uploadRes = await fetch(`/api/v2/files/company/${targetCompanyId}/upload`, {
    method: 'POST',
    credentials: 'include',
    body: formData,
  });

  if (!uploadRes.ok) {
    const errText = await uploadRes.text();
    throw new Error(errText || 'Upload failed');
  }

  const result = await uploadRes.json();
  const docId = result.id || result.document?.id || result.data?.id;
  if (!docId) throw new Error('No document ID returned');

  // Step 2: Link to the DD requirement (this also sets secondaryCategory on the backend)
  const linkRes = await fetch(`/api/v2/due-diligence/company/${targetCompanyId}/requirement/${requirementId}/status`, {
    method: 'PUT',
    credentials: 'include',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      status: 'in_progress',
      documentId: docId,
    }),
  });

  if (!linkRes.ok) throw new Error('Failed to link document');

  // Step 3: Refresh
  queryClient.invalidateQueries({ queryKey: ['/api/v2/due-diligence/company', targetCompanyId] });
  queryClient.invalidateQueries({ queryKey: ['/api/v2/files/company', targetCompanyId] });
};
```
