```tsx
const totalFees = useMemo(() => {
  if (!modelDetails) return 0;
  const offerPrice = toNum(modelDetails.totalPurchasePrice) || toNum(modelDetails.enterpriseValue);
  if (offerPrice <= 0) return 0;
  const fee = (val: any, type: any) => type === 'fixed' ? toNum(val) : offerPrice * toNum(val) / 100;
  return fee(modelDetails.feeLegalDdPercent, modelDetails.feeLegalDdType)
       + fee(modelDetails.feeAccountingDdPercent, modelDetails.feeAccountingDdType)
       + fee(modelDetails.feeBrokerPercent, modelDetails.feeBrokerType)
       + fee(modelDetails.feeLoanArrangementPercent, modelDetails.feeLoanArrangementType)
```

**C) Value Creation Bridge** — the hero visual (shown in chart view):

This is a **waterfall/bridge chart** — a classic PE visualisation showing how entry equity transforms into total return through value creation levers.

**Bridge data array** (build in `useMemo`):

```tsx
const bridgeData = useMemo(() => {
  const m = metrics;
  const items: Array<{
    name: string;
    value: number;
    isTotal: boolean;
    color: string;
  }> = [
    { name: 'Entry\nEquity', value: m.entryEquity, isTotal: true, color: 'hsl(215, 20%, 65%)' },
    { name: 'EBITDA\nGrowth', value: m.ebitdaGrowthValue, isTotal: false, color: 'hsl(142, 71%, 45%)' },
    { name: 'Multiple\nExpansion', value: m.multipleExpansionValue, isTotal: false, color: 'hsl(199, 89%, 48%)' },
    { name: 'Debt\nPaydown', value: m.debtPaydownValue, isTotal: false, color: 'hsl(262, 83%, 58%)' },
    { name: 'Cash\nDistributions', value: m.cashDistributionsValue, isTotal: false, color: 'hsl(45, 93%, 47%)' },
    { name: 'Fees &\nCosts', value: -m.feesDeducted, isTotal: false, color: 'hsl(0, 72%, 51%)' },
    { name: 'Total\nReturn', value: m.totalReturn, isTotal: true, color: 'hsl(160, 84%, 39%)' },
  ];

  // Build waterfall positions: each bar needs a base (invisible) and visible portion
  let runningTotal = 0;
  return items.map(item => {
    if (item.isTotal) {
      // Total bars start from 0
      const result = {
        ...item,
        base: 0,
        barValue: item.value,
        displayValue: item.value,
      };
      if (item.name.includes('Entry')) {
        runningTotal = item.value;
      }
      return result;
    } else {
      // Incremental bars float from the running total
      const base = item.value >= 0 ? runningTotal : runningTotal + item.value;
      const barValue = Math.abs(item.value);
      const result = {
        ...item,
        base,
        barValue,
        displayValue: item.value,
      };
      runningTotal += item.value;
      return result;
    }
  });
}, [metrics]);
```

**Rendering the bridge chart:**

Use Recharts `BarChart` with two stacked `Bar` components sharing the same `stackId`:
1. **Base bar** (invisible): `stackId="bridge"`, `fill="transparent"`, `dataKey="base"`
2. **Value bar** (visible): `stackId="bridge"`, `dataKey="barValue"`, with per-cell colouring using `<Cell>` elements

```tsx
<BarChart data={bridgeData} margin={{ top: 25, right: 10, left: 10, bottom: 5 }}>
  <CartesianGrid strokeDasharray="3 3" className="stroke-muted" vertical={false} />
  <XAxis
    dataKey="name"
    tick={{ fontSize: 11 }}
    tickLine={false}
    axisLine={false}
    interval={0}
  />
  <YAxis
    tick={{ fontSize: 11 }}
    tickLine={false}
    axisLine={false}
    tickFormatter={(v: number) => {
      if (Math.abs(v) >= 1000000) return `£${(v / 1000000).toFixed(1)}M`;
      if (Math.abs(v) >= 1000) return `£${(v / 1000).toFixed(0)}K`;
      return `£${v}`;
    }}
    width={70}
  />
  <ChartTooltip
    content={({ active, payload }) => {
      if (!active || !payload?.length) return null;
      const data = payload[0]?.payload;
      if (!data) return null;
      return (
        <div className="rounded-lg border bg-background px-3 py-2 shadow-xl">
          <p className="text-sm font-medium">{data.name?.replace('\n', ' ')}</p>
          <p className="text-xs font-mono mt-1">
            {data.displayValue >= 0 ? '+' : ''}{formatCurrency(data.displayValue)}
          </p>
        </div>
      );
    }}
  />
  <Bar dataKey="base" stackId="bridge" fill="transparent" isAnimationActive={false} />
  <Bar dataKey="barValue" stackId="bridge" isAnimationActive={false} radius={[4, 4, 0, 0]}>
    {bridgeData.map((entry, index) => (
      <Cell key={index} fill={entry.color} />
    ))}
    <LabelList
      dataKey="displayValue"
      position="top"
      formatter={(value: number) => {
        if (Math.abs(value) >= 1000000) return `£${(value / 1000000).toFixed(1)}M`;
        if (Math.abs(value) >= 1000) return `£${(value / 1000).toFixed(0)}K`;
        return formatCurrency(value);
      }}
      className="text-[10px] fill-muted-foreground font-mono"
    />
  </Bar>
</BarChart>
```

### Edge Cases

- If `metrics.irr` is `NaN` or `Infinity` or extremely large (>10 i.e. 1000%), display "N/A" instead of the number
- If `metrics.moic` is 0, display "0.00x"
- If `metrics.paybackPeriodYears` is 0 or equals the hold period exactly, show ">{holdPeriod}yrs" in red
- If `multipleExpansionValue` is negative (multiple contraction), the bridge bar should still render correctly as a downward step — the `base` calculation handles this automatically since `item.value` will be negative
