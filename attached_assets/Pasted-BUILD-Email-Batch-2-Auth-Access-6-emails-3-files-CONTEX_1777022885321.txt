STEP 3: Add a private escapeHtml method to the EmailService class in server/email.ts

Inside the EmailService class in server/email.ts, immediately before the closing brace of the class (the final `}` before `export const emailService = new EmailService();`), insert:

  private escapeHtml(str: string): string {
    return String(str ?? '')
      .replace(/&/g, '&amp;')
      .replace(/</g, '&lt;')
      .replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;')
      .replace(/'/g, '&#39;');
  }

  async sendPortalAccessLink(
    contactEmail: string,
    contactName: string,
    companyName: string,
    portalUrl: string,
    customMessage?: string,
    expiresAt?: Date
  ): Promise<boolean> {
    const safeContactName = this.escapeHtml(contactName);
    const safeCompanyName = this.escapeHtml(companyName);
    const expiryText = expiresAt
      ? `This link expires on ${expiresAt.toLocaleDateString('en-GB')}.`
      : 'This link has a limited validity period.';

    const blocks: EmailBlock[] = [
      { type: 'paragraph', text: `Hello ${safeContactName},` },
      { type: 'paragraph', text: `You have been given secure access to the document portal for the <strong style="color:#FFFFFF;">${safeCompanyName}</strong> transaction.` },
    ];

    if (customMessage && customMessage.trim()) {
      blocks.push({
        type: 'info_box',
        box: { tone: 'info', body: this.escapeHtml(customMessage.trim()) },
      });
    }

    blocks.push({ type: 'cta', cta: { label: 'Access document portal', href: portalUrl } });
    blocks.push({ type: 'paragraph', text: `${expiryText} Please do not share this link.` });

    const html = renderBrandedEmail({
      preheader: `Secure document portal access for the ${companyName} transaction.`,
      eyebrow: 'Portal access',
      heading: 'Your document portal is ready',
      blocks,
      footerLine: 'You are receiving this because secure document portal access was granted to your email address.',
    });

    return this.sendEmail({
      to: contactEmail,
      subject: `Document Portal Access — ${companyName}`,
      html,
    });
  }

  async sendWelcomeEmail(params: {
    to: string;
    firstName: string;
    planKey: string;
    magicLink: string;
    freeMonths: number;
    trialDays: number;
  }): Promise<SendInvitationResult> {
    const { to, firstName, planKey, magicLink, freeMonths, trialDays } = params;
    const safeFirstName = escapeHtml(firstName);

    const planLabels: Record<string, string> = {
      CORE: 'Core', PROFESSIONAL: 'Professional', EXPERT: 'Expert', ENTERPRISE: 'Enterprise',
    };
    const planLabel = planLabels[planKey] || planKey;

    const planDescriptions: Record<string, string> = {
      CORE: '5 target companies &middot; Deal Modeller &middot; AI Screening',
      PROFESSIONAL: '20 target companies &middot; Full Expert Analysis &middot; Data Room &middot; Team Seats',
      EXPERT: '100 target companies &middot; All features &middot; Priority support',
    };
    const planDesc = planDescriptions[planKey] || '';

    const formatDate = (d: Date): string => {
      const months = ['January','February','March','April','May','June','July','August','September','October','November','December'];
      return `${d.getDate()} ${months[d.getMonth()]} ${d.getFullYear()}`;
    };

    const blocks: EmailBlock[] = [
      { type: 'paragraph', text: `Hi ${safeFirstName},` },
      { type: 'paragraph', text: `Your <strong style="color:#FFFFFF;">${planLabel}</strong> plan is now active.` },
    ];

    if (freeMonths > 0) {
      const endDate = new Date();
      endDate.setMonth(endDate.getMonth() + freeMonths);
      blocks.push({
        type: 'info_box',
        box: {
          tone: 'success',
          body: `Your <strong>${freeMonths} months free</strong> period starts today. No charge until <strong>${formatDate(endDate)}</strong>.`,
        },
      });
    } else if (trialDays > 0) {
      const endDate = new Date(Date.now() + trialDays * 24 * 60 * 60 * 1000);
      blocks.push({
        type: 'info_box',
        box: {
          tone: 'info',
          body: `Your <strong>${trialDays} day free trial</strong> has started. You will not be charged until <strong>${formatDate(endDate)}</strong>.`,
        },
      });
    }

    blocks.push({ type: 'cta', cta: { label: 'Access Valoxi', href: magicLink } });
    blocks.push({ type: 'paragraph', text: 'This link expires in 72 hours. If it expires, visit the login page to request a new one.' });

    if (planDesc) {
      blocks.push({ type: 'divider' });
      blocks.push({
        type: 'info_box',
        box: {
          tone: 'info',
          title: `${planLabel} plan`,
          body: planDesc,
        },
      });
    }

    const html = renderBrandedEmail({
      preheader: `Your Valoxi ${planLabel} plan is active. Click inside to access your account.`,
      eyebrow: 'Welcome',
      heading: `Welcome to Valoxi, ${safeFirstName}`,
      blocks,
      footerLine: 'You are receiving this because a Valoxi account was created for your email address.',
    });

    const success = await (legacyEmailService as any).sendEmail({
      to,
      subject: 'Welcome to Valoxi — click here to access your account',
      html,
    });
    return { success, message: success ? 'Welcome email sent' : 'Failed to send welcome email' };
  }

  async requestReset(email: string, baseUrl: string): Promise<void> {
    const user = await usersRepository.getByEmail(email);
    if (!user) return;

    await passwordResetRepository.deleteUserTokens(user.id);
    const tokenRow = await passwordResetRepository.createToken(user.id);

    const resetUrl = `${baseUrl}/reset-password?token=${tokenRow.token}`;

    const blocks: EmailBlock[] = [
      { type: 'paragraph', text: 'Hi,' },
      { type: 'paragraph', text: 'We received a request to reset the password for your Valoxi account. Click the button below to choose a new password.' },
      { type: 'cta', cta: { label: 'Reset password', href: resetUrl } },
      { type: 'paragraph', text: 'This link expires in <strong style="color:#FFFFFF;">1 hour</strong>. If you did not request a password reset, you can safely ignore this email.' },
    ];

    const html = renderBrandedEmail({
      preheader: 'A request was made to reset your Valoxi password.',
      eyebrow: 'Password reset',
      heading: 'Reset your Valoxi password',
      blocks,
      footerLine: 'You are receiving this because a password reset was requested for your Valoxi account.',
    });

    await (legacyEmailService as any).sendEmail({
