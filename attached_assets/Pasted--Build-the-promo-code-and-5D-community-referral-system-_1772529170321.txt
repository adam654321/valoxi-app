typescriptfunction computeFreeMonthsElapsed(freeMonthsStartedAt: Date): number {
  const now = new Date()
  const months = (now.getFullYear() - freeMonthsStartedAt.getFullYear()) * 12
    + (now.getMonth() - freeMonthsStartedAt.getMonth())
  return Math.floor(months)
}
