Step 2: delete inside a single transaction
Only run this after step 1 looks clean. The transaction guarantees that if any step fails (e.g. an unexpected FK), nothing commits.
sqlBEGIN;

-- Capture the deletion set into a temp table so we don't recompute it
CREATE TEMP TABLE _to_delete AS
  SELECT id, email, clerk_user_id
    FROM users
   WHERE user_company_id = 5
     AND id NOT IN (1, 2);

-- Show what we captured (sanity)
SELECT * FROM _to_delete;

-- Delete in dependency order
DELETE FROM user_invitations
 WHERE user_company_id = 5
   AND email IN (SELECT email FROM _to_delete);

DELETE FROM user_invitations
 WHERE invited_by IN (SELECT id FROM _to_delete);

DELETE FROM user_sessions
 WHERE user_id IN (SELECT id FROM _to_delete);

DELETE FROM user_company_memberships
 WHERE user_id IN (SELECT id FROM _to_delete);

DELETE FROM token_usage
 WHERE user_id IN (SELECT id FROM _to_delete);

DELETE FROM password_reset_tokens
 WHERE user_id IN (SELECT id FROM _to_delete);

DELETE FROM magic_links
 WHERE user_id IN (SELECT id FROM _to_delete);

-- Finally remove the users themselves
DELETE FROM users
 WHERE id IN (SELECT id FROM _to_delete);

-- Confirm clean state before commit
SELECT id, email, first_name, last_name FROM users WHERE user_company_id = 5 ORDER BY id;
SELECT id, email, invitation_status FROM user_invitations WHERE user_company_id = 5 ORDER BY id;
SELECT clerk_user_id FROM _to_delete WHERE clerk_user_id IS NOT NULL;

-- If the two SELECTs above show only Adam (id=1) and Viv (id=2) and
-- the invitations list contains only the two legacy 'unknown' rows
-- (or is empty), commit. Otherwise ROLLBACK and tell me.
COMMIT;
