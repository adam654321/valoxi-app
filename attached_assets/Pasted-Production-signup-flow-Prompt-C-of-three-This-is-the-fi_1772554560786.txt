Step 5 — Cohort 2 Mid-Free-Period Upgrade with Pro-Rata Credit:
In server/routes/billing.routes.ts, in POST /api/v2/billing/upgrade, add this block before the normal upgrade logic:
typescriptif (subscription.freeMonthsTotal > 0 && subscription.freeMonthsStartedAt) {
  const now = new Date()
  const startedAt = new Date(subscription.freeMonthsStartedAt)
  const elapsedMs = now.getTime() - startedAt.getTime()
  const elapsedMonths = elapsedMs / (1000 * 60 * 60 * 24 * 30.44)
  const unusedMonths = Math.max(0, subscription.freeMonthsTotal - elapsedMonths)
  const creditAmountPence = Math.round(unusedMonths * 5025) // £50.25 in pence

  if (creditAmountPence > 0) {
    await stripe.customers.update(subscription.stripeCustomerId, {
      balance: -creditAmountPence
    })
  }

  // Look up campaign for postFreePeriodCouponId
  const promoCode = subscription.promoCodeId
    ? await db.query.promoCodes.findFirst({ where: eq(promoCodes.id, subscription.promoCodeId) })
    : null
  const campaign = promoCode
    ? await db.query.promoCodeCampaigns.findFirst({ where: eq(promoCodeCampaigns.id, promoCode.campaignId) })
    : null

  if (campaign?.postFreePeriodCouponId) {
    await stripe.customers.update(subscription.stripeCustomerId, {
      coupon: campaign.postFreePeriodCouponId
    })
  }

  await db.update(subscriptions).set({
    freeMonthsTotal: 0,
    freeMonthsStartedAt: null,
    communityDiscountCouponId: campaign?.postFreePeriodCouponId || null,
  }).where(eq(subscriptions.userCompanyId, userCompanyId))

  await billingRepo.logEvent({
    userCompanyId,
    eventType: 'subscription.upgraded_from_free_period',
    metadata: { unusedMonths, creditAmountPence, newPlanKey: planKey }
  })
}
