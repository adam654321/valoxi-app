CONTEXT: We are building the Deal Documents feature. Step 5 is the final Sprint A step — creating template .docx files with {{placeholder}} syntax that docx-templates can populate, and seeding them as system templates in the database.

We have three uploaded example templates at /mnt/user-data/uploads/:
- NDA_.docx
- Offer_Letter.docx  
- HOT_Including_Indemnity_Letter_Template__3_.docx

We need to create NEW .docx files based on these with {{placeholder}} tags replacing the bracketed text, then store them in data/templates/ and insert records into deal_document_templates.

TASK: Create a seed script server/scripts/seed-deal-document-templates.ts that:

1. Ensures data/templates/ directory exists
2. For each template, creates a clean .docx with {{placeholder}} syntax using the docx library (NOT docx-templates — we use the `docx` npm package to CREATE template files)
3. Inserts records into deal_document_templates

First: npm install docx (the document creation library, separate from docx-templates)

Then create the seed script:
```typescript
import { db } from '../db';
import { dealDocumentTemplates } from '@shared/schema/deal-documents';
import { eq, and } from 'drizzle-orm';
import fs from 'fs/promises';
import path from 'path';

const TEMPLATES_DIR = path.join(process.cwd(), 'data', 'templates');

/**
 * For the initial version, we store the uploaded template files directly
 * and define a placeholder mapping that documents which text patterns
 * in the templates correspond to which data context fields.
 * 
 * The template engine will be enhanced to handle the actual placeholder
 * replacement. For now, we copy the uploaded files and register them.
 */

interface TemplateConfig {
  templateName: string;
  documentType: string;
  sourceFile: string;
  placeholderMap: Record<string, string>;
  sections?: Array<{ key: string; label: string; required: boolean }>;
}

const TEMPLATES: TemplateConfig[] = [
  {
    templateName: 'Standard NDA',
    documentType: 'nda',
    sourceFile: 'NDA_.docx',
    placeholderMap: {
      // Maps placeholder keys in the template to DocumentDataContext fields
      'buyerCompanyName': 'buyerCompanyName',
      'sellerCompanyName': 'sellerCompanyName',
      'sellerCompanyNumber': 'sellerCompanyNumber',
      'sellerAddress': 'sellerAddress',
      'currentDate': 'currentDate',
      'buyerSignatoryName': 'buyerSignatoryName',
    },
  },
  {
    templateName: 'Standard Offer Letter',
    documentType: 'offer_letter',
    sourceFile: 'Offer_Letter.docx',
    placeholderMap: {
      'buyerCompanyName': 'buyerCompanyName',
      'sellerCompanyName': 'sellerCompanyName',
      'sellerContactName': 'sellerContactName',
      'sellerAddress': 'sellerAddress',
      'currentDate': 'currentDate',
      'buyerSignatoryName': 'buyerSignatoryName',
      'averageEbitda': 'averageEbitda',
      'ebitdaMultiple': 'ebitdaMultiple',
      'enterpriseValue': 'enterpriseValue',
      'offerPrice': 'offerPrice',
      'propertyValue': 'propertyValue',
      'excessCash': 'excessCash',
      'primaryDebt': 'primaryDebt',
      'otherDebt': 'otherDebt',
      'corpTaxDue': 'corpTaxDue',
      'workingCapitalAdj': 'workingCapitalAdj',
      'dayOnePayment': 'dayOnePayment',
      'deferredTotal': 'deferredTotal',
    },
  },
  {
    templateName: 'Standard Heads of Terms',
    documentType: 'heads_of_terms',
    sourceFile: 'HOT_Including_Indemnity_Letter_Template__3_.docx',
    placeholderMap: {
      'buyerCompanyName': 'buyerCompanyName',
      'sellerCompanyName': 'sellerCompanyName',
      'sellerContactName': 'sellerContactName',
      'sellerAddress': 'sellerAddress',
      'currentDate': 'currentDate',
      'buyerSignatoryName': 'buyerSignatoryName',
      'offerPrice': 'offerPrice',
      'dayOnePayment': 'dayOnePayment',
      'deferredTotal': 'deferredTotal',
      'exclusivityPeriodDays': 'exclusivityPeriodDays',
      'nonCompetePeriodYears': 'nonCompetePeriodYears',
      'nonSolicitPeriodYears': 'nonSolicitPeriodYears',
      'warrantyClaimPeriodMonths': 'warrantyClaimPeriodMonths',
      'warrantyMinClaim': 'warrantyMinClaim',
    },
    sections: [
      { key: 'proposed_acquisition', label: 'The Proposed Acquisition', required: true },
      { key: 'price', label: 'Price', required: true },
      { key: 'assumptions', label: 'Assumptions', required: false },
      { key: 'conditions', label: 'Conditions', required: false },
      { key: 'due_diligence', label: 'Due Diligence', required: true },
      { key: 'share_purchase_agreement', label: 'Share Purchase Agreement', required: false },
      { key: 'timetable', label: 'Timetable and Negotiations', required: true },
      { key: 'exclusivity', label: 'Exclusivity', required: true },
      { key: 'costs', label: 'Costs', required: false },
      { key: 'law', label: 'Law', required: true },
    ],
  },
];

async function seedTemplates() {
  console.log('Seeding deal document templates...');
  
  await fs.mkdir(TEMPLATES_DIR, { recursive: true });
  
  for (const config of TEMPLATES) {
    // Check if system template already exists for this type
    const [existing] = await db.select().from(dealDocumentTemplates)
      .where(
        and(
          eq(dealDocumentTemplates.documentType, config.documentType),
          eq(dealDocumentTemplates.isSystem, true)
        )
      );
    
    if (existing) {
      console.log(`  ✓ ${config.templateName} already exists (id: ${existing.id})`);
      continue;
    }
    
    // Copy the source file to templates directory
    const sourcePath = path.join(process.cwd(), 'attached_assets', config.sourceFile);
    const destPath = path.join(TEMPLATES_DIR, `system_${config.documentType}.docx`);
    
    try {
      await fs.copyFile(sourcePath, destPath);
      console.log(`  Copied ${config.sourceFile} → ${destPath}`);
    } catch (err) {
      // Try alternate location
      const altPath = path.join('/mnt/user-data/uploads', config.sourceFile);
      try {
        await fs.copyFile(altPath, destPath);
        console.log(`  Copied from uploads: ${config.sourceFile} → ${destPath}`);
      } catch (err2) {
        console.error(`  ✗ Could not find source file: ${config.sourceFile}`);
        console.error(`    Tried: ${sourcePath} and ${altPath}`);
        continue;
      }
    }
    
    // Insert template record
    const [record] = await db.insert(dealDocumentTemplates).values({
      templateName: config.templateName,
      documentType: config.documentType,
      templateFileUrl: destPath,
      placeholderMap: config.placeholderMap,
      sections: config.sections || null,
      isSystem: true,
      isActive: true,
    }).returning();
    
    console.log(`  ✓ Created ${config.templateName} (id: ${record.id})`);
  }
  
  console.log('Template seeding complete.');
}

// Run if called directly
seedTemplates()
  .then(() => process.exit(0))
  .catch((err) => {
    console.error('Seed failed:', err);
    process.exit(1);
  });

export { seedTemplates };
```

IMPORTANT NOTES:
- The source files might be in `attached_assets/` (Replit's upload directory) or the original upload paths. The script tries both locations.
- For the initial version, we're copying the uploaded templates as-is and storing the placeholder mapping as metadata. The template engine will be enhanced to do the actual text replacement using this mapping.
- The HoT template has a `sections` array — these are the toggleable sections the buyer can include/exclude before sending.
- The script is idempotent — running it twice won't create duplicates.
- After creating the file, RUN the seed script: npx tsx server/scripts/seed-deal-document-templates.ts
