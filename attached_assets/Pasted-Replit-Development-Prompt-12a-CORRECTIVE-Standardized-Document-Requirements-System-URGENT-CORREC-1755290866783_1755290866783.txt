const standardRequirementsManager = {
  // Initialize standard requirements for any target company
  initializeStandardRequirements: async (targetCompanyId, createdBy) => {
    // Verify permissions
    const user = await getUserWithPermissions(createdBy);
    if (!user.can_create_deals && !user.can_manage_company) {
      throw new Error('Insufficient permissions to initialize document requirements');
    }
    
    // Call the database function to create standard requirements
    await db.query('SELECT initialize_standard_requirements($1)', [targetCompanyId]);
    
    // Log activity
    await logActivity(targetCompanyId, createdBy, 'standard_requirements_initialized', 'Initialized 29 standard document requirements (6 Stage 1 + 23 Stage 2)');
    
    return {
      stage_1_requirements: 6,
      stage_2_requirements: 23,
      total_requirements: 29,
      message: 'Standard document requirements initialized successfully'
    };
  },
  
  // Get requirements for a company (simplified)
  getCompanyRequirements: async (targetCompanyId, stage = null) => {
    let requirementsQuery = `
      SELECT 
        cdr.*,
        CASE 
          WHEN cdr.due_date < NOW() THEN 'overdue'
          WHEN cdr.requirement_status = 'submitted' THEN 'submitted'
          WHEN cdr.requirement_status = 'approved' THEN 'approved'
          WHEN cdr.requirement_status = 'waived' THEN 'waived'
          ELSE 'outstanding'
        END as display_status
      FROM company_document_requirements cdr
      WHERE cdr.target_company_id = $1
    `;
    
    const queryParams = [targetCompanyId];
    
    if (stage !== null) {
      requirementsQuery += ' AND cdr.collection_stage = $2';
      queryParams.push(stage);
    }
    
    requirementsQuery += ' ORDER BY cdr.collection_stage ASC, cdr.priority_level DESC, cdr.requirement_name ASC';
    
    const result = await db.query(requirementsQuery, queryParams);
    return result.rows;
  }
};

const initializeStandardRequirements = async (targetCompanyId) => {
  try {
    const response = await fetch(`/api/companies/${targetCompanyId}/requirements/initialize-standard`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      }
    });
    
    const result = await response.json();
    
    if (response.ok) {
      showNotification(`Initialized ${result.total_requirements} standard requirements`, 'success');
      location.reload(); // Refresh to show requirements
    } else {
      showNotification(result.error || 'Failed to initialize requirements', 'error');
    }
    
  } catch (error) {
    console.error('Requirements initialization failed:', error);
    showNotification('Failed to initialize requirements', 'error');
  }
};
