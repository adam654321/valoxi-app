typescriptconst session = event.data.object as Stripe.Checkout.Session
const { userCompanyId, planKey, billingInterval, promoCodeId, isNewSignup } = session.metadata

if (isNewSignup === 'true') {
  const customerEmail = session.customer_details?.email || session.customer_email

  const existingUser = await db.query.users.findFirst({
    where: eq(users.email, customerEmail)
  })

  if (existingUser) {
    // EMAIL EXISTS — link payment to existing account
    await billingService.activateSubscription({
      userCompanyId: existingUser.userCompanyId,
      stripeCustomerId: session.customer as string,
      stripeSubscriptionId: session.subscription as string,
      planKey,
      billingInterval,
    })
    await emailService.sendPaymentLinkedEmail({
      to: customerEmail,
      firstName: existingUser.firstName || 'there',
      planKey,
    })

  } else {
    // NEW EMAIL — create full account
    const [userCompany] = await db.insert(userCompanies).values({
      name: session.customer_details?.name || 'My Company',
      onboardingCompleted: false,
    }).returning()

    const tempPassword = crypto.randomBytes(16).toString('hex')
    const passwordHash = await bcrypt.hash(tempPassword, 12)
    const nameParts = (session.customer_details?.name || '').split(' ')

    const [user] = await db.insert(users).values({
      email: customerEmail,
      firstName: nameParts[0] || '',
      lastName: nameParts.slice(1).join(' ') || '',
      passwordHash,
      userCompanyId: userCompany.id,
      onboardingCompleted: false,
    }).returning()

    const trialEndsAt = promoCodeId ? null : new Date(Date.now() + 7 * 24 * 60 * 60 * 1000)
    await db.insert(subscriptions).values({
      userCompanyId: userCompany.id,
      stripeCustomerId: session.customer as string,
      stripeSubscriptionId: session.subscription as string,
      planKey,
      status: 'active',
      billingInterval,
      trialEndsAt,
    })

    if (promoCodeId) {
      await promoCodeService.claimCode(promoCodeId, userCompany.id, session.customer as string)
    }

    await seedDefaultPipelineStages(userCompany.id)

    const magicLink = await magicLinkService.generateMagicLink(user.id, 'welcome')

    await emailService.sendWelcomeEmail({
      to: customerEmail,
      firstName: nameParts[0] || 'there',
      planKey,
      magicLink,
      freeMonths: promoCodeId ? 3 : 0,
      trialDays: promoCodeId ? 0 : 7,
    })

    console.log('[webhook] New user created via Stripe checkout:', customerEmail, planKey)
  }

} else {
  // EXISTING USER PATH — upgrade/activation already built, keep as is
}
