  /**
   * Create and generate an Offer Letter from a Deal Model.
   * Pulls all commercial terms, formats them, populates the template.
   */
  async createOfferFromDealModel(params: {
    targetCompanyId: number;
    userCompanyId: number;
    dealModelId: number;
    portalTokenId: number;
    createdBy: number;
  }) {
    // Check if there's already a sent/viewed offer — if so, this becomes a new version
    const existingOffer = await this.getLatestActiveDocument(params.targetCompanyId, 'offer_letter');
    
    let doc;
    if (existingOffer && ['sent', 'viewed', 'countered', 'rejected'].includes(existingOffer.status)) {
      // Create new version
      doc = await this.createNewVersion(existingOffer.id, params.createdBy, {
        dealModelId: params.dealModelId,
      });
    } else {
      // Get the system offer template
      const { dealDocumentTemplateService } = await import('./deal-document-template.service');
      const template = await dealDocumentTemplateService.getSystemTemplate('offer_letter');
      
      doc = await this.createDocument({
        targetCompanyId: params.targetCompanyId,
        userCompanyId: params.userCompanyId,
        documentType: 'offer_letter',
        createdBy: params.createdBy,
        dealModelId: params.dealModelId,
        portalTokenId: params.portalTokenId,
        templateId: template?.id,
      });
    }
    
    // Build the offer data context from the deal model
    const offerData = await this.buildOfferDataContext(params.dealModelId, params.targetCompanyId, params.userCompanyId, params.createdBy);
    
    // Generate the document file
    await this.generateDocumentFile(doc.id, offerData);
    
    return doc;
  }

  /**
   * Build the data context for an offer letter from a deal model.
   * Pulls calculated values and formats them for the template.
   */
  private async buildOfferDataContext(
    dealModelId: number,
    targetCompanyId: number,
    userCompanyId: number,
    createdBy: number
  ): Promise<Record<string, any>> {
    // Get base context (buyer/seller details, date)
    const { dealDocumentTemplateService } = await import('./deal-document-template.service');
    const baseContext = await dealDocumentTemplateService.buildDataContext(
      targetCompanyId, userCompanyId, createdBy, dealModelId
    );
    
    // Get the deal model with its funding lines
    const [model] = await db.select().from(dealModels)
      .where(eq(dealModels.id, dealModelId));
    
    if (!model) throw new Error('Deal model not found');
    
    // Get funding lines for this model
    const { dealModelFundingLines } = await import('@shared/schema/deal-models');
    const fundingLines = await db.select().from(dealModelFundingLines)
      .where(eq(dealModelFundingLines.dealModelId, dealModelId));
    
    // Calculate the full deal model to get computed values
    const { calculateFullDealModel } = await import('@shared/calculators/deal-model.calc');
    
    // Build calculator input from the model — match what DealModeller.tsx does
    const modelInput: Record<string, any> = {};
    
    // Copy all model fields that the calculator needs
    const fieldsToCopy = [
      'ebitdaMultiple', 'averagingYears',
      'propertyValue', 'plusExcessCash',
      'minusPrimaryDebt', 'minusOtherDebt', 'minusCorpTaxDue', 'minusWorkingCapitalAdj',
      'feeLegalDd', 'feeAccountingDd', 'feeBroker', 'feeLoanArrangement', 'feeDealMaker',
      'feeLegalDdType', 'feeAccountingDdType', 'feeBrokerType', 'feeLoanArrangementType', 'feeDealMakerType',
      'dayOneEquity', 'dayOneUnsecuredLoan', 'dayOneVendorDeposit',
    ];
    
    for (const field of fieldsToCopy) {
      if ((model as any)[field] !== undefined && (model as any)[field] !== null) {
        modelInput[field] = (model as any)[field];
      }
    }
    
    // Add EBITDA data from expert snapshot if available
    const snapshot = (model as any).expertDataSnapshot;
    if (snapshot?.qoe?.adjusted_ebitda_by_year) {
      modelInput.ebitdaByYear = snapshot.qoe.adjusted_ebitda_by_year;
    }
    
    // Add funding lines
    modelInput.fundingLines = fundingLines.map(fl => ({
      id: fl.id,
      name: fl.name,
      type: fl.type,
      amount: fl.amount,
      interestRate: fl.interestRate,
      termMonths: fl.termMonths,
      arrangementFeePercent: fl.arrangementFeePercent,
    }));
    
    // Run calculator
    let calculations: any = {};
    try {
      calculations = calculateFullDealModel(modelInput);
    } catch (err) {
      console.error('[DEAL_DOCS] Calculator error:', err);
    }
    
    // Format currency helper
    const fmt = (val: any) => {
      const num = Number(val) || 0;
      return new Intl.NumberFormat('en-GB', {
        style: 'currency', currency: 'GBP',
        minimumFractionDigits: 0, maximumFractionDigits: 0,
      }).format(num);
    };
    
    // Build offer-specific context
    const offerPrice = calculations.offerPrice || calculations.equityValue || 0;
    const enterpriseValue = calculations.enterpriseValue || 0;
    const excessCash = Number(model.plusExcessCash) || 0;
    const dayOnePayment = Number(model.dayOneEquity) || 0;
    
    // Calculate deferred payments
    // Deferred = offer price - day one payment (simplified)
    const deferredTotal = Math.max(0, offerPrice - dayOnePayment - excessCash);
    
    // Check if model has deferred schedule data
    const deferredYears = 5; // Default to 5 years
    const deferredInstalment = deferredTotal > 0 ? Math.round(deferredTotal / deferredYears) : 0;
    
    return {
      ...baseContext,
      // Financial values — formatted as currency
      offerPrice: fmt(offerPrice),
      enterpriseValue: fmt(enterpriseValue),
      excessCash: fmt(excessCash),
      dayOnePayment: fmt(dayOnePayment),
      deferredTotal: fmt(deferredTotal),
      deferredInstalment: fmt(deferredInstalment),
      
      // Raw values for the documentData snapshot
      _raw: {
        offerPrice,
        enterpriseValue,
        excessCash,
        dayOnePayment,
        deferredTotal,
        deferredInstalment,
        deferredYears,
        averageEbitda: calculations.averageEbitda || 0,
        ebitdaMultiple: Number(model.ebitdaMultiple) || 0,
      },
    };
  }
