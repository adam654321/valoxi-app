-- Create document requirement submissions tracking table
CREATE TABLE IF NOT EXISTS document_requirement_submissions (
  id SERIAL PRIMARY KEY,
  company_requirement_id INTEGER REFERENCES company_document_requirements(id),
  external_document_id INTEGER REFERENCES external_document_submissions(id),
  target_company_id INTEGER REFERENCES target_companies(id),
  target_contact_id INTEGER REFERENCES target_contacts(id),
  
  -- Submission Details
  submitted_at TIMESTAMP DEFAULT NOW(),
  submission_method VARCHAR(50) DEFAULT 'portal_upload', -- portal_upload, email_attachment, manual_entry
  
  -- Document Information
  original_filename VARCHAR(255),
  file_size BIGINT,
  file_type VARCHAR(50),
  
  -- Status Tracking
  is_active BOOLEAN DEFAULT TRUE,
  replaced_by_submission_id INTEGER REFERENCES document_requirement_submissions(id),
  
  -- Validation
  is_validated BOOLEAN DEFAULT FALSE,
  validated_by INTEGER REFERENCES users(id),
  validated_at TIMESTAMP,
  validation_notes TEXT,
  
  created_at TIMESTAMP DEFAULT NOW()
);

-- Create stage 1 progress tracking view
CREATE OR REPLACE VIEW stage_1_progress AS
SELECT 
  tc.id as target_company_id,
  tc.company_name,
  COUNT(cdr.id) as total_requirements,
  COUNT(CASE WHEN cdr.requirement_status IN ('submitted', 'approved') THEN 1 END) as completed_requirements,
  COUNT(CASE WHEN cdr.requirement_status = 'submitted' THEN 1 END) as submitted_count,
  COUNT(CASE WHEN cdr.requirement_status = 'approved' THEN 1 END) as approved_count,
  ROUND(
    (COUNT(CASE WHEN cdr.requirement_status IN ('submitted', 'approved') THEN 1 END)::DECIMAL / 
     COUNT(cdr.id)::DECIMAL) * 100, 1
  ) as completion_percentage,
  CASE 
    WHEN COUNT(CASE WHEN cdr.requirement_status IN ('submitted', 'approved') THEN 1 END) = COUNT(cdr.id) THEN 'complete'
    WHEN COUNT(CASE WHEN cdr.requirement_status IN ('submitted', 'approved') THEN 1 END) > 0 THEN 'in_progress'
    ELSE 'not_started'
  END as overall_status,
  MAX(cdr.updated_at) as last_activity
FROM target_companies tc
LEFT JOIN company_document_requirements cdr ON tc.id = cdr.target_company_id AND cdr.collection_stage = 1
GROUP BY tc.id, tc.company_name;

-- Create indexes for performance
CREATE INDEX IF NOT EXISTS idx_doc_req_submissions_company ON document_requirement_submissions(target_company_id);
CREATE INDEX IF NOT EXISTS idx_doc_req_submissions_requirement ON document_requirement_submissions(company_requirement_id);
CREATE INDEX IF NOT EXISTS idx_doc_req_submissions_active ON document_requirement_submissions(is_active, submitted_at);

const documentTrackingManager = {
  // Get Stage 1 requirements with progress for portal display
  getStage1RequirementsForPortal: async (portalToken) => {
    // Verify portal token and get company information
    const tokenQuery = `
      SELECT 
        pat.*,
        tc.id as target_company_id,
        tc.company_name,
        tc.user_company_id
      FROM portal_access_tokens pat
      JOIN target_companies tc ON pat.target_company_id = tc.id
      WHERE pat.access_token = $1 
      AND pat.is_active = true 
      AND pat.expires_at > NOW()
    `;
    
    const tokenResult = await db.query(tokenQuery, [portalToken]);
    if (tokenResult.rows.length === 0) {
      throw new Error('Invalid or expired portal token');
    }
    
    const tokenData = tokenResult.rows[0];
    
    // Get Stage 1 requirements with submission status
    const requirementsQuery = `
      SELECT 
        cdr.*,
        COALESCE(drs.submission_count, 0) as submitted_documents,
        drs.latest_submission_date,
        drs.latest_filename,
        CASE 
          WHEN cdr.requirement_status IN ('submitted', 'approved') THEN 'completed'
          WHEN drs.submission_count > 0 THEN 'submitted'
          ELSE 'pending'
        END as display_status
      FROM company_document_requirements cdr
      LEFT JOIN (
        SELECT 
          company_requirement_id,
          COUNT(*) as submission_count,
          MAX(submitted_at) as latest_submission_date,
          MAX(original_filename) as latest_filename
        FROM document_requirement_submissions 
        WHERE is_active = true
        GROUP BY company_requirement_id
      ) drs ON cdr.id = drs.company_requirement_id
      WHERE cdr.target_company_id = $1 
      AND cdr.collection_stage = 1
      ORDER BY cdr.priority_level DESC, cdr.requirement_name ASC
    `;
    
    const requirements = await db.query(requirementsQuery, [tokenData.target_company_id]);
    
    // Calculate overall progress
    const totalRequirements = requirements.rows.length;
    const completedRequirements = requirements.rows.filter(req => 
      req.requirement_status === 'submitted' || req.requirement_status === 'approved'
    ).length;
    
    const progressPercentage = totalRequirements > 0 
      ? Math.round((completedRequirements / totalRequirements) * 100)
      : 0;
    
    return {
      company_name: tokenData.company_name,
      total_requirements: totalRequirements,
      completed_requirements: completedRequirements,
      progress_percentage: progressPercentage,
      overall_status: completedRequirements === totalRequirements ? 'complete' : 
                     completedRequirements > 0 ? 'in_progress' : 'not_started',
      requirements: requirements.rows,
      portal_expires_at: tokenData.expires_at
    };
  },
  
  // Submit document for specific requirement via portal
  submitDocumentForRequirement: async (portalToken, requirementId, documentData, uploadedFile) => {
    // Verify portal token
    const tokenQuery = `
      SELECT pat.*, tc.id as target_company_id
      FROM portal_access_tokens pat
      JOIN target_companies tc ON pat.target_company_id = tc.id
      WHERE pat.access_token = $1 AND pat.is_active = true AND pat.expires_at > NOW()
    `;
    
    const tokenResult = await db.query(tokenQuery, [portalToken]);
    if (tokenResult.rows.length === 0) {
      throw new Error('Invalid or expired portal token');
    }
    
    const tokenData = tokenResult.rows[0];
    
    // Verify requirement belongs to this company and is Stage 1
    const requirementQuery = `
      SELECT * FROM company_document_requirements 
      WHERE id = $1 AND target_company_id = $2 AND collection_stage = 1
    `;
    
    const requirement = await db.query(requirementQuery, [requirementId, tokenData.target_company_id]);
    if (requirement.rows.length === 0) {
      throw new Error('Invalid requirement or requirement not found');
    }
    
    const requirementData = requirement.rows[0];
    
    // Check if requirement already completed
    if (requirementData.requirement_status === 'approved') {
      throw new Error('This requirement has already been completed and approved');
    }
    
    // Create external document submission record
    const documentQuery = `
      INSERT INTO external_document_submissions (
        target_company_id, target_contact_id, portal_token_id,
        original_filename, file_path, file_size, file_type,
        submission_notes, processing_status
      ) VALUES ($1, $2, $3, $4, $5, $6, $7, $8, $9)
      RETURNING *
    `;
    
    const documentValues = [
      tokenData.target_company_id,
      tokenData.target_contact_id,
      tokenData.id,
      uploadedFile.originalname,
      uploadedFile.path,
      uploadedFile.size,
      uploadedFile.mimetype,
      documentData.submission_notes || '',
      'uploaded'
    ];
    
    const documentResult = await db.query(documentQuery, documentValues);
    const document = documentResult.rows[0];
    
    // Create requirement submission tracking record
    const submissionQuery = `
      INSERT INTO document_requirement_submissions (
        company_requirement_id, external_document_id, target_company_id, target_contact_id,
        original_filename, file_size, file_type
      ) VALUES ($1, $2, $3, $4, $5, $6, $7)
      RETURNING *
    `;
    
    const submissionValues = [
      requirementId,
      document.id,
      tokenData.target_company_id,
      tokenData.target_contact_id,
      uploadedFile.originalname,
      uploadedFile.size,
      uploadedFile.mimetype
    ];
    
    const submissionResult = await db.query(submissionQuery, submissionValues);
    
    // Update requirement status
    await db.query(`
      UPDATE company_document_requirements SET
        requirement_status = 'submitted',
        submitted_document_id = $2,
        submitted_at = NOW(),
        submitted_by_contact_id = $3,
        submitted_document_count = submitted_document_count + 1,
        latest_submission_date = NOW()
      WHERE id = $1
    `, [requirementId, document.id, tokenData.target_contact_id]);
    
    // Update stage progression
    await updateStage1Progression(tokenData.target_company_id);
    
    // Log activity
    await logPortalActivity(tokenData.id, 'document_submitted', `Document submitted for: ${requirementData.requirement_name}`, document.id);
    
    // Create internal notification
    await notifyInternalTeam(
      tokenData.target_company_id,
      'stage1_document_submitted',
      `Stage 1 document submitted: ${requirementData.requirement_name}`,
      document.id
    );
    
    return {
      submission_id: submissionResult.rows[0].id,
      document_id: document.id,
      requirement_name: requirementData.requirement_name,
      filename: uploadedFile.originalname,
      submitted_at: new Date().toISOString()
    };
  },
  
  // Update Stage 1 progression and check for completion
  updateStage1Progression: async (targetCompanyId) => {
    // Get current Stage 1 progress
    const progressQuery = `
      SELECT * FROM stage_1_progress WHERE target_company_id = $1
    `;
    
    const progress = await db.query(progressQuery, [targetCompanyId]);
    if (progress.rows.length === 0) return;
    
    const progressData = progress.rows[0];
    
    // Update document stage progression table
    await db.query(`
      UPDATE document_stage_progression SET
        stage_1_completion_percentage = $2,
        stage_1_mandatory_completed = $3,
        completed_requirements = $4,
        overall_completion_percentage = $5,
        can_progress_to_offer = $3,
        updated_at = NOW()
      WHERE target_company_id = $1
    `, [
      targetCompanyId,
      progressData.completion_percentage,
      progressData.overall_status === 'complete',
      progressData.completed_requirements,
      progressData.completion_percentage
    ]);
    
    // Check if Stage 1 is complete
    if (progressData.overall_status === 'complete') {
      // Mark Stage 1 as completed
      await db.query(`
        UPDATE document_stage_progression SET
          stage_1_completed_at = NOW(),
          stage_1_status = 'completed'
        WHERE target_company_id = $1
      `, [targetCompanyId]);
      
      // Create notification for Stage 1 completion
      await notifyInternalTeam(
        targetCompanyId,
        'stage1_completed',
        'Stage 1 document collection completed - ready for analysis',
        null
      );
      
      // Log completion activity
      await logActivity(targetCompanyId, null, 'stage1_completed', 'All Stage 1 documents submitted and ready for analysis');
      
      return { stage1_complete: true, ready_for_analysis: true };
    }
    
    return { stage1_complete: false, progress: progressData.completion_percentage };
  },
  
  // Get document progress for internal dashboard
  getDocumentProgressDashboard: async (userCompanyId) => {
    const dashboardQuery = `
      SELECT 
        tc.id,
        tc.company_name,
        tc.industry,
        tc.stage as pipeline_stage,
        sp.total_requirements,
        sp.completed_requirements,
        sp.completion_percentage,
        sp.overall_status,
        sp.last_activity,
        CASE 
          WHEN sp.overall_status = 'complete' THEN '✅ Complete'
          WHEN sp.overall_status = 'in_progress' THEN '🔄 In Progress'
          ELSE '⏳ Not Started'
        END as status_display,
        COUNT(DISTINCT pat.id) as active_portals
      FROM target_companies tc
      LEFT JOIN stage_1_progress sp ON tc.id = sp.target_company_id
      LEFT JOIN portal_access_tokens pat ON tc.id = pat.target_company_id AND pat.is_active = true
      WHERE tc.user_company_id = $1
      AND tc.is_active = true
      GROUP BY tc.id, tc.company_name, tc.industry, tc.stage, sp.total_requirements, 
               sp.completed_requirements, sp.completion_percentage, sp.overall_status, sp.last_activity
      ORDER BY sp.completion_percentage DESC, tc.company_name ASC
    `;
    
    const result = await db.query(dashboardQuery, [userCompanyId]);
    
    // Calculate summary statistics
    const companies = result.rows;
    const totalCompanies = companies.length;
    const completedCompanies = companies.filter(c => c.overall_status === 'complete').length;
    const inProgressCompanies = companies.filter(c => c.overall_status === 'in_progress').length;
    const notStartedCompanies = companies.filter(c => c.overall_status === 'not_started').length;
    
    const averageCompletion = totalCompanies > 0 
      ? Math.round(companies.reduce((sum, c) => sum + (parseFloat(c.completion_percentage) || 0), 0) / totalCompanies)
      : 0;
    
    return {
      summary: {
        total_companies: totalCompanies,
        completed_companies: completedCompanies,
        in_progress_companies: inProgressCompanies,
        not_started_companies: notStartedCompanies,
        average_completion: averageCompletion
      },
      companies: companies
    };
  },
  
  // Validate submitted document (internal user action)
  validateSubmittedDocument: async (documentId, validationData, validatedBy) => {
    // Verify permissions
    const user = await getUserWithPermissions(validatedBy);
    if (!user.can_edit_all_deals && !user.can_manage_company) {
      throw new Error('Insufficient permissions to validate documents');
    }
    
    // Get document submission details
    const submissionQuery = `
      SELECT 
        drs.*,
        cdr.requirement_name,
        cdr.target_company_id,
        eds.original_filename
      FROM document_requirement_submissions drs
      JOIN company_document_requirements cdr ON drs.company_requirement_id = cdr.id
      JOIN external_document_submissions eds ON drs.external_document_id = eds.id
      WHERE drs.id = $1
    `;
    
    const submission = await db.query(submissionQuery, [documentId]);
    if (submission.rows.length === 0) {
      throw new Error('Document submission not found');
    }
    
    const submissionData = submission.rows[0];
    
    // Update validation status
    await db.query(`
      UPDATE document_requirement_submissions SET
        is_validated = $2,
        validated_by = $3,
        validated_at = NOW(),
        validation_notes = $4
      WHERE id = $1
    `, [documentId, validationData.is_approved, validatedBy, validationData.validation_notes || '']);
    
    // Update requirement status based on validation
    const newStatus = validationData.is_approved ? 'approved' : 'submitted';
    await db.query(
      'UPDATE company_document_requirements SET requirement_status = $1 WHERE id = $2',
      [newStatus, submissionData.company_requirement_id]
    );
    
    // Update stage progression
    await updateStage1Progression(submissionData.target_company_id);
    
    // Log validation activity
    const activityMessage = `Document ${validationData.is_approved ? 'approved' : 'rejected'}: ${submissionData.requirement_name}`;
    await logActivity(submissionData.target_company_id, validatedBy, 'document_validated', activityMessage);
    
    return {
      document_id: documentId,
      requirement_name: submissionData.requirement_name,
      validation_status: validationData.is_approved ? 'approved' : 'rejected',
      validated_at: new Date().toISOString()
    };
  }
};

const logPortalActivity = async (portalTokenId, activityType, activityDescription, relatedDocumentId = null) => {
  const activityQuery = `
    INSERT INTO portal_activity_log (
      portal_token_id, activity_type, activity_description, 
      related_document_id, activity_timestamp
    ) VALUES ($1, $2, $3, $4, NOW())
  `;
  
  await db.query(activityQuery, [portalTokenId, activityType, activityDescription, relatedDocumentId]);
};

const notifyInternalTeam = async (targetCompanyId, notificationType, notificationMessage, relatedId = null) => {
  // Get all users assigned to this deal
  const teamQuery = `
    SELECT DISTINCT u.id
    FROM users u
    JOIN user_deal_assignments uda ON u.id = uda.user_id
    WHERE uda.target_company_id = $1 AND uda.is_active = true
  `;
  
  const team = await db.query(teamQuery, [targetCompanyId]);
  
  // Create notifications for all team members
  for (const member of team.rows) {
    await createTeamNotification(
      member.id,
      notificationType,
      notificationMessage,
      notificationMessage,
      targetCompanyId,
      relatedId
    );
  }
};

        return;
      }
      
      // Disable form during submission
      submitBtn.disabled = true;
      submitBtn.textContent = '⏳ Uploading...';
      
      try {
        const formData = new FormData();
        formData.append('document', fileInput.files[0]);
        formData.append('requirement_id', requirementId);
        formData.append('submission_notes', notesTextarea.value);
        
        const response = await fetch(`/api/portal/${getPortalToken()}/documents/submit`, {
          method: 'POST',
          body: formData
        });
        
        const result = await response.json();
        
        if (response.ok) {
          showNotification(`Document submitted successfully: ${result.filename}`, 'success');
          
          // Refresh page to show updated progress
          setTimeout(() => {
            location.reload();
          }, 1500);
          
        } else {
          throw new Error(result.error || 'Upload failed');
        }
        
      } catch (error) {
        console.error('Document upload failed:', error);
        showNotification(`Upload failed: ${error.message}`, 'error');
        
        // Re-enable form
        submitBtn.disabled = false;
        submitBtn.textContent = `📤 Submit Document`;
      }
    });
  });
});

const formatFileSize = (bytes) => {
  if (bytes === 0) return '0 Bytes';
  const k = 1024;
  const sizes = ['Bytes', 'KB', 'MB', 'GB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
};

const getPortalToken = () => {

    }
  });

  });
};
