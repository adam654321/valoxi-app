FILE 1 (NEW): server/lib/nylas-state-token.ts
  Small helper module for state-token sign/verify, separated from
  the service so it can be unit-tested independently:

    import jwt from 'jsonwebtoken';
    
    type NylasStateClaims = {
      userId: number;
      userCompanyId: number;
      provider: 'google' | 'microsoft';
    };
    
    type SignedNylasState = NylasStateClaims & { iat: number; exp: number };
    
    const STATE_TOKEN_TTL_SECONDS = 300; // 5 minutes
    
    function getSecret(): string {
      const secret = process.env.NYLAS_STATE_SECRET;
      if (!secret) {
        throw new Error('NYLAS_STATE_SECRET environment variable is required');
      }
      if (secret.length < 32) {
        throw new Error('NYLAS_STATE_SECRET must be at least 32 characters');
      }
      return secret;
    }
    
    export function signNylasState(claims: NylasStateClaims): string {
      return jwt.sign(claims, getSecret(), {
        expiresIn: STATE_TOKEN_TTL_SECONDS,
        algorithm: 'HS256',
      });
    }
    
    export function verifyNylasState(token: string): SignedNylasState {
      try {
        const decoded = jwt.verify(token, getSecret(), {
          algorithms: ['HS256'],
        });
        if (typeof decoded === 'string') {
          throw new Error('Unexpected JWT payload type');
        }
        return decoded as SignedNylasState;
      } catch (err) {
        if (err instanceof jwt.TokenExpiredError) {
          throw new Error('OAuth state token has expired (older than 5 minutes)');
        }
        if (err instanceof jwt.JsonWebTokenError) {
          throw new Error('OAuth state token is invalid');
        }
        throw err;
      }
    }

    async getAuthUrl(provider: Provider, stateToken: string): Promise<string> {
      this.getClient(); // throws NylasServiceNotConfiguredError if not configured
      
      const scopes = this.getScopesForProvider(provider);
      
      const params = new URLSearchParams({
        client_id: config.nylas.clientId,
        redirect_uri: config.nylas.redirectUri,
        response_type: 'code',
        provider,
        scope: scopes,
        state: stateToken,
      });
      
      // Google needs access_type=offline for refresh tokens.
      // Microsoft uses offline_access as a scope instead — already
      // included in the Microsoft scope string.
      if (provider === 'google') {
        params.set('access_type', 'offline');
      }
      
      // Do NOT include `prompt` — Nylas v3 rejects prompt=consent.
      
      return `${config.nylas.apiUri}/v3/connect/auth?${params.toString()}`;
    }
    
    private getScopesForProvider(provider: Provider): string {
      if (provider === 'google') {
        return 'https://www.googleapis.com/auth/gmail.send openid https://www.googleapis.com/auth/userinfo.email';
      }
      if (provider === 'microsoft') {
        // NOT YET TESTED end-to-end (CC handoff Section 5). Verify
        // during Build 2 dev V-check by completing a Microsoft auth
        // flow and confirming a grant row lands.
        return 'Mail.Send offline_access openid email User.Read';
      }
      throw new Error(`Unknown provider: ${provider}`);
    }

  exchangeCodeForGrant: POST to /v3/connect/token.

    async exchangeCodeForGrant(code: string): Promise<{
      grantId: string;
      email: string;
      provider: Provider;
      scope: string[];
    }> {
      this.getClient();
      
      const response = await fetch(`${config.nylas.apiUri}/v3/connect/token`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Accept': 'application/json',
        },
        body: JSON.stringify({
          client_id: config.nylas.clientId,
          client_secret: config.nylas.apiKey, // NYLAS_API_KEY doubles as client_secret
          code,
          redirect_uri: config.nylas.redirectUri,
          grant_type: 'authorization_code',
        }),
      });
      
      if (!response.ok) {
        const errorBody = await response.text();
        throw new Error(
          `Nylas code exchange failed (${response.status}): ${errorBody.slice(0, 500)}`
        );
      }
      
      const data = await response.json() as {
        grant_id: string;
        email: string;
        provider: Provider;
        scope: string;
      };
      
      // scope arrives as space-separated string per OAuth convention
      const scopeArray = data.scope ? data.scope.split(' ').filter(Boolean) : [];
      
      return {
        grantId: data.grant_id,
        email: data.email,
        provider: data.provider,
        scope: scopeArray,
      };
    }

FILE 3 (NEW): server/routes/nylas.routes.ts

  Two endpoints. Mount path TBD by T2/T5 findings — likely under
  /api/v2/nylas with the callback exempted from CSRF.

    import { Router, Request, Response } from 'express';
    import { z } from 'zod';
    import { nylasService, NylasServiceNotConfiguredError } from '../services/nylas.service';
    import { nylasRepository } from '../repositories/nylas.repository';
    import { signNylasState, verifyNylasState } from '../lib/nylas-state-token';
    
    const router = Router();
    
    // POST /api/v2/nylas/auth/initiate
    // Authenticated. CSRF-protected (matches /api/v2/* convention).
    // Body: { provider: 'google' | 'microsoft' }
    // Returns: { authUrl: string }
    
    const initiateBodySchema = z.object({
      provider: z.enum(['google', 'microsoft']),
    });
    
    router.post('/auth/initiate', async (req: Request, res: Response) => {
      try {
        const { provider } = initiateBodySchema.parse(req.body);
        
        const userId = req.session.userId;
        const userCompanyId = req.session.userCompanyId;
        if (!userId || !userCompanyId) {
          return res.status(401).json({ success: false, error: 'Not authenticated' });
        }
        
        const stateToken = signNylasState({ userId, userCompanyId, provider });
        const authUrl = await nylasService.getAuthUrl(provider, stateToken);
        
        return res.json({ success: true, data: { authUrl } });
      } catch (err) {
        if (err instanceof NylasServiceNotConfiguredError) {
          return res.status(503).json({ success: false, error: err.message });
        }
        if (err instanceof z.ZodError) {
          return res.status(400).json({ success: false, error: 'Invalid request body', details: err.errors });
        }
        console.error('[NYLAS] auth/initiate failed:', err);
        return res.status(500).json({ success: false, error: 'Failed to start OAuth flow' });
      }
    });
    
    // GET /api/v2/nylas/auth/callback
    // UNAUTHENTICATED. Called by Nylas (third party) on completion of
    // OAuth flow. Must be exempt from CSRF middleware (no JS-originated
    // request, no token to send).
    // Query: ?code=...&state=...&error=...&error_description=...
    
    router.get('/auth/callback', async (req: Request, res: Response) => {
      try {
        // Nylas may redirect with ?error=... if user cancelled or provider rejected
        if (req.query.error) {
          const errorDescription = String(req.query.error_description ?? req.query.error);
          console.warn(`[NYLAS] auth/callback received provider error: ${errorDescription}`);
          return res.redirect(`/settings/connected-accounts?error=${encodeURIComponent(errorDescription)}`);
        }
        
        const code = req.query.code;
        const state = req.query.state;
        
        if (typeof code !== 'string' || typeof state !== 'string') {
          return res.redirect('/settings/connected-accounts?error=missing_code_or_state');
        }
        
        let claims;
        try {
          claims = verifyNylasState(state);
        } catch (err) {
          console.warn('[NYLAS] auth/callback state verification failed:', err);
          return res.redirect('/settings/connected-accounts?error=invalid_state');
        }
        
        const { grantId, email, provider, scope } = await nylasService.exchangeCodeForGrant(code);
        
        // Idempotency: if a grant already exists with this grantId, do not double-insert
        const existing = await nylasRepository.getGrantByGrantId(grantId);
        if (!existing) {
          await nylasRepository.createGrant({
            userId: claims.userId,
            userCompanyId: claims.userCompanyId,
            grantId,
            email,
            provider,
            scopes: scope,
          });
        }
        
        return res.redirect(`/settings/connected-accounts?connected=true&email=${encodeURIComponent(email)}`);
      } catch (err) {
        console.error('[NYLAS] auth/callback failed:', err);
        return res.redirect('/settings/connected-accounts?error=callback_failed');
      }
    });
    
    export default router;
