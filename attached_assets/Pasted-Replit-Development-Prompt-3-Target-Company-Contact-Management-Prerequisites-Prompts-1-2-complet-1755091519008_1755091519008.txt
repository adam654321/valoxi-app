// Add contact to target company
const addContact = async (targetCompanyId, contactData, userId) => {
  // Verify user owns this target company
  const targetCompany = await getTargetCompanyById(targetCompanyId, userId);
  if (!targetCompany) {
    throw new Error('Target company not found or access denied');
  }
  
  // If this is marked as primary, unset other primary contacts
  if (contactData.is_primary) {
    await db.query(
      'UPDATE target_contacts SET is_primary = FALSE WHERE target_company_id = $1',
      [targetCompanyId]
    );
  }
  
  const query = `
    INSERT INTO target_contacts 
    (target_company_id, first_name, last_name, title, email, phone, is_primary, notes)
    VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
    RETURNING *
  `;
  
  const values = [
    targetCompanyId,
    contactData.first_name,
    contactData.last_name,
    contactData.title,
    contactData.email,
    contactData.phone,
    contactData.is_primary || false,
    contactData.notes
  ];
  
  return await db.query(query, values);
};

// Get contacts for target company (with security check)
const getContactsForCompany = async (targetCompanyId, userId) => {
  const query = `
    SELECT tc.* FROM target_contacts tc
    JOIN target_companies t ON tc.target_company_id = t.id
    JOIN users u ON t.user_company_id = u.user_company_id
    WHERE tc.target_company_id = $1 AND u.id = $2
    ORDER BY tc.is_primary DESC, tc.last_name ASC
  `;
  
  return await db.query(query, [targetCompanyId, userId]);
};
Contact Validation
const validateContact = (data) => {
  const errors = [];
  
  if (!data.first_name || data.first_name.trim().length < 1) {
    errors.push('First name is required');
  }
  
  if (!data.last_name || data.last_name.trim().length < 1) {
    errors.push('Last name is required');
  }
  
  if (data.email && !isValidEmail(data.email)) {
    errors.push('Please enter a valid email address');
  }
  
  if (data.phone && data.phone.length > 0 && data.phone.length < 10) {
    errors.push('Please enter a valid phone number');
  }
  
  return errors;
};

// Middleware to verify target company access
const verifyTargetCompanyAccess = async (req, res, next) => {
  const targetCompanyId = req.params.id;
  const userId = req.user.id;
  
  const query = `
    SELECT t.* FROM target_companies t
    JOIN users u ON t.user_company_id = u.user_company_id
    WHERE t.id = $1 AND u.id = $2
  `;
  
  const result = await db.query(query, [targetCompanyId, userId]);
  
  if (result.rows.length === 0) {
    return res.status(404).json({ error: 'Target company not found' });
  }
  
  req.targetCompany = result.rows[0];
  next();
};
