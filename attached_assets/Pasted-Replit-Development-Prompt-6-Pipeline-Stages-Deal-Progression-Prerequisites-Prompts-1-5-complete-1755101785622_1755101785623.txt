Initial Contact - First outreach made to target company

-- Create stage history table
CREATE TABLE stage_history (
  id SERIAL PRIMARY KEY,
  target_company_id INTEGER REFERENCES target_companies(id),
  from_stage VARCHAR(50),
  to_stage VARCHAR(50),
  changed_by INTEGER REFERENCES users(id),
  changed_at TIMESTAMP DEFAULT NOW(),
  notes TEXT
);

-- Create stage definitions table
CREATE TABLE pipeline_stages (
  id SERIAL PRIMARY KEY,
  stage_name VARCHAR(50) UNIQUE NOT NULL,
  stage_order INTEGER NOT NULL,
  description TEXT,
  is_active BOOLEAN DEFAULT TRUE
);

4. Stage Update Implementation
const updateCompanyStage = async (companyId, newStage, userId, notes = '') => {
  // Verify user has access to this company
  const company = await getTargetCompanyById(companyId, userId);
  if (!company) {
    throw new Error('Company not found or access denied');
  }
  
  const oldStage = company.stage;
  
  // Don't update if stage is the same
  if (oldStage === newStage) {
    return { success: true, message: 'Stage unchanged' };
  }
  
  try {
    // Start transaction
    await db.query('BEGIN');
    
    // Update company stage
    await db.query(
      'UPDATE target_companies SET stage = $1, stage_updated_at = NOW(), stage_updated_by = $2 WHERE id = $3',
      [newStage, userId, companyId]
    );
    
    // Record stage history
    await db.query(
      'INSERT INTO stage_history (target_company_id, from_stage, to_stage, changed_by, notes) VALUES ($1, $2, $3, $4, $5)',
      [companyId, oldStage, newStage, userId, notes]
    );
    
    // Commit transaction
    await db.query('COMMIT');
    
    return {
      success: true,
      message: `Stage updated from "${oldStage}" to "${newStage}"`,
      oldStage,
      newStage
    };
    
  } catch (error) {
    await db.query('ROLLBACK');
    throw error;
  }
};
5. Pipeline Overview Dashboard
const getPipelineOverview = async (userId) => {
  const user = await getUserById(userId);
  
  const query = `
    SELECT 
      ps.stage_name,
      ps.stage_order,
      COUNT(tc.id) as company_count,
      AVG(EXTRACT(days FROM (NOW() - tc.stage_updated_at))) as avg_days_in_stage
    FROM pipeline_stages ps
    LEFT JOIN target_companies tc ON ps.stage_name = tc.stage 
      AND tc.user_company_id = $1
    WHERE ps.is_active = true
    GROUP BY ps.stage_name, ps.stage_order
    ORDER BY ps.stage_order
  `;
  
  const stageStats = await db.query(query, [user.user_company_id]);
  
  // Get recent stage changes
  const recentChanges = await db.query(`
    SELECT 
      sh.*, 
      tc.company_name,
      u.first_name || ' ' || u.last_name as changed_by_name
    FROM stage_history sh
    JOIN target_companies tc ON sh.target_company_id = tc.id
    JOIN users u ON sh.changed_by = u.id
    WHERE tc.user_company_id = $1
    ORDER BY sh.changed_at DESC
    LIMIT 10
  `, [user.user_company_id]);
  
  return {
    stageStats: stageStats.rows,
    recentChanges: recentChanges.rows,
    totalCompanies: stageStats.rows.reduce((sum, stage) => sum + parseInt(stage.company_count), 0)
  };
};

Stage Update JavaScript
const updateStage = async () => {
  const newStage = document.getElementById('stage-select').value;
  const notes = document.getElementById('stage-notes').value;
  const companyId = getCurrentCompanyId(); // Get from URL or page data
  
  try {
    const response = await fetch(`/api/target-companies/${companyId}/stage`, {
      method: 'PUT',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        stage: newStage,
        notes: notes
      })
    });
    
    const result = await response.json();
    
    if (result.success) {
      // Update UI
      updateStageDisplay(result);
      
      // Clear notes
      document.getElementById('stage-notes').value = '';
      
      // Show success message
      showNotification('Stage updated successfully', 'success');
      
      // Refresh stage history if displayed
      refreshStageHistory();
    } else {
      showNotification('Failed to update stage', 'error');
    }
    
  } catch (error) {
    console.error('Stage update failed:', error);
    showNotification('Error updating stage', 'error');
  }
};

const updateStageDisplay = (result) => {
  document.getElementById('stage-updated-date').textContent = new Date().toLocaleDateString();
  document.getElementById('days-in-stage').textContent = '0';
  
  // Update any other UI elements that show current stage
  const stageElements = document.querySelectorAll('.current-stage-display');
  stageElements.forEach(el => {
    el.textContent = result.newStage;
  });
};
