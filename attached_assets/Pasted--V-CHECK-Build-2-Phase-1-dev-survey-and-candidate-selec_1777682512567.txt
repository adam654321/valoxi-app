QUERIES TO RUN

1) All chase-up schedules on dev with token-level state:

SELECT
  cs.id AS schedule_id,
  cs.portal_token_id,
  cs.target_company_id,
  cs.submission_deadline,
  CURRENT_DATE - cs.submission_deadline AS days_overdue,
  cs.is_active,
  cs.is_completed,
  cs.completion_reason,
  pat.chase_up_status,
  pat.escalation_level,
  pat.chase_email_count
FROM chase_up_schedules cs
INNER JOIN portal_access_tokens pat ON cs.portal_token_id = pat.id
ORDER BY cs.id;

2) Requirements breakdown for every targetCompanyId that appears in (1):

SELECT
  cdr.target_company_id,
  cdr.collection_stage,
  COUNT(*) AS total_rows,
  COUNT(*) FILTER (WHERE cdr.is_mandatory IS NOT FALSE) AS mandatory_rows,
  COUNT(*) FILTER (WHERE cdr.is_mandatory IS NOT FALSE AND cdr.requirement_status IN ('submitted','approved')) AS mandatory_terminal,
  COUNT(*) FILTER (WHERE cdr.is_mandatory IS NOT FALSE AND cdr.requirement_status = 'outstanding') AS mandatory_outstanding,
  COUNT(*) FILTER (WHERE cdr.is_mandatory IS NOT FALSE AND cdr.requirement_status = 'rejected') AS mandatory_rejected
FROM company_document_requirements cdr
WHERE cdr.target_company_id IN (SELECT DISTINCT target_company_id FROM chase_up_schedules)
GROUP BY cdr.target_company_id, cdr.collection_stage
ORDER BY cdr.target_company_id, cdr.collection_stage;

3) Quick check for any target_company_id in (1) that has ZERO rows in company_document_requirements:

SELECT DISTINCT cs.target_company_id
FROM chase_up_schedules cs
WHERE NOT EXISTS (
  SELECT 1 FROM company_document_requirements cdr
  WHERE cdr.target_company_id = cs.target_company_id
);
