const { Pool } = require('pg');
const PizZip = require('pizzip');
const Docxtemplater = require('docxtemplater');
const fs = require('fs');
const pool = new Pool({ connectionString: process.env.DATABASE_URL });
pool.query(\`
  SELECT dd.generated_file_url
  FROM deal_documents dd
  JOIN target_companies tc ON dd.target_company_id = tc.id
  WHERE tc.company_name ILIKE '%macinnes%'
  AND dd.document_type = 'offer_letter'
  ORDER BY dd.created_at DESC
  LIMIT 1
\`).then(r => {
  const path = r.rows[0]?.generated_file_url;
  console.log('Generated file path:', path);
  if (path && fs.existsSync(path)) {
    const content = fs.readFileSync(path, 'binary');
    const zip = new PizZip(content);
    const doc = new Docxtemplater(zip, { paragraphLoop: true, linebreaks: true });
    const text = doc.getFullText();
    console.log('First 300 chars:', text.substring(0, 300));
  } else {
    console.log('File does not exist at that path');
  }
  pool.end();
}).catch(e => { console.error(e); pool.end(); });
