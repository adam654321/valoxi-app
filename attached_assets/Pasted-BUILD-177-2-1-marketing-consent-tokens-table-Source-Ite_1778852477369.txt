    BEGIN;

    CREATE TABLE marketing_consent_tokens (
      id SERIAL PRIMARY KEY,
      token VARCHAR(64) NOT NULL,
      user_id INTEGER REFERENCES users(id) ON DELETE CASCADE,
      waitlist_id UUID REFERENCES waitlist(id) ON DELETE CASCADE,
      email VARCHAR(255) NOT NULL,
      resend_message_id VARCHAR(255),
      sent_at TIMESTAMP NOT NULL DEFAULT NOW(),
      consent_recorded_at TIMESTAMP,
      ip_address INET,
      expires_at TIMESTAMP NOT NULL,
      CONSTRAINT marketing_consent_tokens_token_unique UNIQUE (token),
      CONSTRAINT marketing_consent_tokens_recipient_check
        CHECK ((user_id IS NOT NULL AND waitlist_id IS NULL) OR (user_id IS NULL AND waitlist_id IS NOT NULL))
    );

    CREATE INDEX marketing_consent_tokens_user_id_idx
      ON marketing_consent_tokens(user_id) WHERE user_id IS NOT NULL;

    CREATE INDEX marketing_consent_tokens_waitlist_id_idx
      ON marketing_consent_tokens(waitlist_id) WHERE waitlist_id IS NOT NULL;

    COMMIT;
