3. Advanced Search Implementation
const searchTargetCompanies = async (userId, searchParams) => {
  const user = await getUserById(userId);
  let query = `
    SELECT t.*, 
           COUNT(tc.id) as contact_count,
           COALESCE(primary_contact.first_name || ' ' || primary_contact.last_name, 'No primary contact') as primary_contact_name
    FROM target_companies t
    LEFT JOIN target_contacts tc ON t.id = tc.target_company_id
    LEFT JOIN target_contacts primary_contact ON t.id = primary_contact.target_company_id AND primary_contact.is_primary = true
    WHERE t.user_company_id = $1
  `;
  
  const params = [user.user_company_id];
  let paramCount = 1;
  
  // Add search conditions
  if (searchParams.q) {
    paramCount++;
    query += ` AND (t.company_name ILIKE $${paramCount} OR t.industry ILIKE $${paramCount} OR t.address ILIKE $${paramCount})`;
    params.push(`%${searchParams.q}%`);
  }
  
  if (searchParams.industry && searchParams.industry !== 'all') {
    paramCount++;
    query += ` AND t.industry = $${paramCount}`;
    params.push(searchParams.industry);
  }
  
  if (searchParams.minRevenue) {
    paramCount++;
    query += ` AND t.annual_revenue >= $${paramCount}`;
    params.push(searchParams.minRevenue);
  }
  
  if (searchParams.maxRevenue) {
    paramCount++;
    query += ` AND t.annual_revenue <= $${paramCount}`;
    params.push(searchParams.maxRevenue);
  }
  
  if (searchParams.minEmployees) {
    paramCount++;
    query += ` AND t.employee_count >= $${paramCount}`;
    params.push(searchParams.minEmployees);
  }
  
  if (searchParams.maxEmployees) {
    paramCount++;
    query += ` AND t.employee_count <= $${paramCount}`;
    params.push(searchParams.maxEmployees);
  }
  
  // Group by for contact count
  query += ` GROUP BY t.id, primary_contact.first_name, primary_contact.last_name`;
  
  // Add sorting
  const sortOptions = {
    'name_asc': 'ORDER BY t.company_name ASC',
    'name_desc': 'ORDER BY t.company_name DESC',
    'date_newest': 'ORDER BY t.created_at DESC',
    'date_oldest': 'ORDER BY t.created_at ASC',
    'revenue_high': 'ORDER BY t.annual_revenue DESC NULLS LAST',
    'revenue_low': 'ORDER BY t.annual_revenue ASC NULLS LAST',
    'employees_high': 'ORDER BY t.employee_count DESC NULLS LAST',
    'employees_low': 'ORDER BY t.employee_count ASC NULLS LAST'
  };
  
  const sortBy = searchParams.sort || 'date_newest';
  query += ` ${sortOptions[sortBy] || sortOptions['date_newest']}`;
  
  return await db.query(query, params);
};

const getFilterOptions = async (userId) => {
  const user = await getUserById(userId);
  
  // Get unique industries for this user's company
  const industries = await db.query(
    'SELECT DISTINCT industry FROM target_companies WHERE user_company_id = $1 AND industry IS NOT NULL ORDER BY industry',
    [user.user_company_id]
  );
  
  // Get revenue ranges
  const revenueStats = await db.query(
    'SELECT MIN(annual_revenue) as min_revenue, MAX(annual_revenue) as max_revenue FROM target_companies WHERE user_company_id = $1 AND annual_revenue IS NOT NULL',
    [user.user_company_id]
  );
  
  // Get employee count ranges
  const employeeStats = await db.query(
    'SELECT MIN(employee_count) as min_employees, MAX(employee_count) as max_employees FROM target_companies WHERE user_company_id = $1 AND employee_count IS NOT NULL',
    [user.user_company_id]
  );
  
  return {
    industries: industries.rows.map(row => row.industry),
    revenueRange: revenueStats.rows[0],
    employeeRange: employeeStats.rows[0]
  };
};

JavaScript for Real-time Search
// Real-time search implementation
let searchTimeout;
const searchInput = document.getElementById('search-input');

searchInput.addEventListener('input', function() {
  clearTimeout(searchTimeout);
  searchTimeout = setTimeout(() => {
    performSearch();
  }, 300); // Debounce search by 300ms
});

const performSearch = async () => {
  const searchParams = {
    q: document.getElementById('search-input').value,
    industry: document.getElementById('industry-filter').value,
    minRevenue: document.getElementById('min-revenue').value,
    maxRevenue: document.getElementById('max-revenue').value,
    minEmployees: document.getElementById('min-employees').value,
    maxEmployees: document.getElementById('max-employees').value,
    sort: document.getElementById('sort-select').value
  };
  
  // Remove empty parameters
  Object.keys(searchParams).forEach(key => {
    if (!searchParams[key] || searchParams[key] === 'all') {
      delete searchParams[key];
    }
  });
  
  try {
    const response = await fetch('/api/target-companies/search?' + new URLSearchParams(searchParams));
    const companies = await response.json();
    
    displayCompanies(companies);
    updateResultsCount(companies.length);
  } catch (error) {
    console.error('Search failed:', error);
  }
};
