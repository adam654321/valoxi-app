2. Database Views for Reporting
-- Create reporting views for better performance
CREATE VIEW pipeline_summary AS
SELECT 
  tc.user_company_id,
  tc.stage,
  COUNT(*) as deal_count,
  AVG(EXTRACT(days FROM (NOW() - tc.stage_updated_at))) as avg_days_in_stage,
  MIN(tc.stage_updated_at) as earliest_stage_entry,
  MAX(tc.stage_updated_at) as latest_stage_entry
FROM target_companies tc
GROUP BY tc.user_company_id, tc.stage;

-- Activity summary view
CREATE VIEW activity_summary AS
SELECT 
  a.user_id,
  tc.user_company_id,
  a.activity_type,
  DATE_TRUNC('month', a.activity_date) as activity_month,
  COUNT(*) as activity_count,
  AVG(a.duration_minutes) as avg_duration
FROM activities a
JOIN target_companies tc ON a.target_company_id = tc.id
GROUP BY a.user_id, tc.user_company_id, a.activity_type, DATE_TRUNC('month', a.activity_date);

-- Deal progression view
CREATE VIEW deal_progression AS
SELECT 
  sh.target_company_id,
  tc.user_company_id,
  sh.from_stage,
  sh.to_stage,
  sh.changed_at,
  LAG(sh.changed_at) OVER (PARTITION BY sh.target_company_id ORDER BY sh.changed_at) as previous_change,
  EXTRACT(days FROM (sh.changed_at - LAG(sh.changed_at) OVER (PARTITION BY sh.target_company_id ORDER BY sh.changed_at))) as days_in_previous_stage
FROM stage_history sh
JOIN target_companies tc ON sh.target_company_id = tc.id;

const getPipelineOverview = async (userId, dateRange = null) => {
  const user = await getUserById(userId);
  
  let dateFilter = '';
  let params = [user.user_company_id];
  
  if (dateRange) {
    dateFilter = 'AND tc.created_at BETWEEN $2 AND $3';
    params.push(dateRange.start, dateRange.end);
  }
  
  // Pipeline distribution
  const pipelineDistribution = await db.query(`
    SELECT 
      stage,
      COUNT(*) as count,
      ROUND(COUNT(*) * 100.0 / SUM(COUNT(*)) OVER (), 2) as percentage
    FROM target_companies tc
    WHERE user_company_id = $1 ${dateFilter}
    GROUP BY stage
    ORDER BY 
      CASE stage
        WHEN 'New Target' THEN 1
        WHEN 'Initial Contact' THEN 2
        WHEN 'Documents Requested' THEN 3
        WHEN 'Under Review' THEN 4
        WHEN 'Offer Stage' THEN 5
        WHEN 'Negotiation' THEN 6
        WHEN 'Due Diligence' THEN 7
        WHEN 'Legal & Accounting' THEN 8
        WHEN 'Completion' THEN 9
        WHEN 'Not Qualified' THEN 10
      END
  `, params);
  
  // Stage velocity (average days in each stage)
  const stageVelocity = await db.query(`
    SELECT 
      from_stage,
      to_stage,
      AVG(days_in_previous_stage) as avg_days,
      COUNT(*) as transition_count
    FROM deal_progression dp
    JOIN target_companies tc ON dp.target_company_id = tc.id
    WHERE tc.user_company_id = $1 
    AND days_in_previous_stage IS NOT NULL
    ${dateFilter.replace('tc.created_at', 'dp.changed_at')}
    GROUP BY from_stage, to_stage
    ORDER BY avg_days DESC
  `, params);
  
  // Conversion rates between stages
  const conversionRates = await db.query(`
    WITH stage_transitions AS (
      SELECT 
        from_stage,
        COUNT(*) as total_exits,
        SUM(CASE WHEN to_stage != 'Not Qualified' THEN 1 ELSE 0 END) as successful_transitions
      FROM deal_progression dp
      JOIN target_companies tc ON dp.target_company_id = tc.id
      WHERE tc.user_company_id = $1 ${dateFilter.replace('tc.created_at', 'dp.changed_at')}
      GROUP BY from_stage
    )
    SELECT 
      from_stage,
      total_exits,
      successful_transitions,
      ROUND(successful_transitions * 100.0 / NULLIF(total_exits, 0), 2) as conversion_rate
    FROM stage_transitions
    ORDER BY conversion_rate DESC
  `, params);
  
  return {
    pipelineDistribution: pipelineDistribution.rows,
    stageVelocity: stageVelocity.rows,
    conversionRates: conversionRates.rows
  };
};
5. Activity Analytics Implementation
const getActivitySummary = async (userId, dateRange = null) => {
  const user = await getUserById(userId);
  
  let dateFilter = '';
  let params = [user.user_company_id];
  
  if (dateRange) {
    dateFilter = 'AND a.activity_date BETWEEN $2 AND $3';
    params.push(dateRange.start, dateRange.end);
  }
  
  // Activity volume by type
  const activityByType = await db.query(`
    SELECT 
      a.activity_type,
      COUNT(*) as count,
      AVG(a.duration_minutes) as avg_duration,
      COUNT(DISTINCT a.target_company_id) as companies_contacted
    FROM activities a
    JOIN target_companies tc ON a.target_company_id = tc.id
    WHERE tc.user_company_id = $1 ${dateFilter}
    GROUP BY a.activity_type
    ORDER BY count DESC
  `, params);
  
  // Activity by user
  const activityByUser = await db.query(`
    SELECT 
      u.first_name || ' ' || u.last_name as user_name,
      COUNT(a.id) as total_activities,
      COUNT(DISTINCT a.target_company_id) as companies_worked,
      AVG(a.duration_minutes) as avg_duration
    FROM users u
    LEFT JOIN activities a ON u.id = a.user_id 
      AND a.activity_date >= CURRENT_DATE - INTERVAL '30 days'
    JOIN target_companies tc ON a.target_company_id = tc.id
    WHERE u.user_company_id = $1 ${dateFilter}
    GROUP BY u.id, u.first_name, u.last_name
    ORDER BY total_activities DESC
  `, params);
  
  // Activity trends (last 12 months)
  const activityTrends = await db.query(`
    SELECT 
      DATE_TRUNC('month', a.activity_date) as month,
      COUNT(*) as activity_count,
      COUNT(DISTINCT a.target_company_id) as active_deals
    FROM activities a
    JOIN target_companies tc ON a.target_company_id = tc.id
    WHERE tc.user_company_id = $1 
    AND a.activity_date >= CURRENT_DATE - INTERVAL '12 months'
    GROUP BY DATE_TRUNC('month', a.activity_date)
    ORDER BY month
  `, params);
  
  return {
    activityByType: activityByType.rows,
    activityByUser: activityByUser.rows,
    activityTrends: activityTrends.rows
  };
};

// Load and display all reports
const loadReports = async (dateRange = '30') => {
  try {
    showLoading();
    
    const [pipelineData, activityData, performanceData] = await Promise.all([
      fetch(`/api/reports/pipeline-overview?range=${dateRange}`).then(r => r.json()),
      fetch(`/api/reports/activity-summary?range=${dateRange}`).then(r => r.json()),
      fetch(`/api/reports/performance-metrics?range=${dateRange}`).then(r => r.json())
    ]);
    
    // Update metrics cards
    updateMetricsCards(performanceData);
    
    // Create charts
    createPipelineChart(pipelineData.pipelineDistribution);
    createActivityTrendsChart(activityData.activityTrends);
    createStageVelocityChart(pipelineData.stageVelocity);
    createUserPerformanceChart(activityData.activityByUser);
    
    hideLoading();
    
  } catch (error) {
    console.error('Failed to load reports:', error);
    showError('Failed to load reports. Please try again.');
  }
};

// Export reports as CSV
const exportReport = async (reportType, format = 'csv') => {
  try {
    const response = await fetch(`/api/reports/export/${reportType}?format=${format}`);
    const blob = await response.blob();
    
    // Create download link
    const url = window.URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${reportType}_report_${new Date().toISOString().split('T')[0]}.${format}`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    window.URL.revokeObjectURL(url);
    
  } catch (error) {
    console.error('Export failed:', error);
    showNotification('Export failed. Please try again.', 'error');
  }
};
