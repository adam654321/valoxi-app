FROM users
WHERE is_active = true
GROUP BY tier;

Query 3: Active users with their RBAC mapping (if any). Adjust column names based on what schema reveals.

  u.is_system_admin,

  u.user_company_id,
  ur.role_name as rbac_role_name,
  ur.can_manage_company as rbac_can_manage_company
FROM users u
LEFT JOIN user_roles ur ON u.user_role = ur.role_name
WHERE u.is_active = true
ORDER BY u.user_company_id, u.id;
