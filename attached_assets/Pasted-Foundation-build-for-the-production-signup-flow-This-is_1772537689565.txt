Step 2 — Magic Link Service:
Create server/services/magic-link.service.ts:
typescriptimport crypto from 'crypto'

// generateMagicLink(userId, type)
// 1. Invalidate existing active links for this user + type:
//    UPDATE magic_links SET invalidated_at = now()
//    WHERE user_id = userId AND type = type AND used_at IS NULL AND invalidated_at IS NULL
// 2. Generate token: crypto.randomBytes(32).toString('hex')
// 3. Insert: userId, token, type, expiresAt = now() + 72 hours
// 4. Return full URL: `${process.env.APP_URL}/auth/magic?token=${token}`
//    If APP_URL not set, fall back to 'http://localhost:5000'

// consumeMagicLink(token)
// 1. Find row WHERE token = token
// 2. If not found → return { valid: false, reason: 'not_found' }
// 3. If invalidatedAt IS NOT NULL → return { valid: false, reason: 'invalidated' }
// 4. If usedAt IS NOT NULL → return { valid: false, reason: 'already_used' }
// 5. If expiresAt < now() → return { valid: false, reason: 'expired' }
// 6. UPDATE magic_links SET used_at = now() WHERE id = row.id
// 7. Return { valid: true, userId: row.userId }

// checkResendRateLimit(email)
// Max 3 requests per email per 1 hour window
// 1. DELETE FROM resend_rate_limits WHERE window_start < now() - interval '1 hour' (clean up old windows)
// 2. Find existing row for this email
// 3. If not found: INSERT new row with requestCount = 1, return { allowed: true, remainingRequests: 2 }
// 4. If found and requestCount >= 3: return { allowed: false, remainingRequests: 0 }
// 5. If found and requestCount < 3: UPDATE requestCount + 1, return { allowed: true, remainingRequests: 3 - newCount }
