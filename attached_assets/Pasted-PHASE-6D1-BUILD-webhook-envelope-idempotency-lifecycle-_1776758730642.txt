Change 3 - writer: logWebhookEvent

File: server/routes/billing.routes.ts (lines 38-56)

Replace the existing logWebhookEvent function with:

async function logWebhookEvent(
  eventType: string,
  stripeEventId: string | null,
  payload: unknown,
  status: string,
  error?: string
) {
  try {
    await db.insert(webhookEvents).values({
      source: 'stripe',
      stripeEventId,
      eventType,
      payload: payload as Record<string, unknown>,
      status,
      error: error ?? null,
      processedAt: new Date(),
    });
  } catch (err) {
    console.error('[Billing] Failed to log webhook event:', err);
  }
}

Change 6 - ON CONFLICT DO NOTHING on subscription_events writer

File: server/repositories/billing.repository.ts (lines 86-89)

Replace the existing logEvent function:

  async logEvent(data: InsertSubscriptionEvent): Promise<SubscriptionEvent> {
    const [row] = await db.insert(subscriptionEvents).values(data).returning();
    return row;
  }

with:

  async logEvent(data: InsertSubscriptionEvent): Promise<SubscriptionEvent | null> {
    if (data.stripeEventId) {
      const [row] = await db
        .insert(subscriptionEvents)
        .values(data)
        .onConflictDoNothing({ target: subscriptionEvents.stripeEventId })
        .returning();
      return row ?? null;
    }
    const [row] = await db.insert(subscriptionEvents).values(data).returning();
    return row;
  }

If your Drizzle version's onConflictDoNothing signature differs (some versions
take no target arg for column-level uniques, since the unique constraint is
auto-detected), drop the target argument and use plain
.onConflictDoNothing().returning(). Pick whichever compiles. Do not change
behaviour beyond that.
