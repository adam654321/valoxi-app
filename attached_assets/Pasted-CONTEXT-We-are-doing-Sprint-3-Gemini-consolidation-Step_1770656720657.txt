      rawResponse: response.text,
      tokenUsage: response.usage ? {
        promptTokens: response.usage.promptTokens,
        candidateTokens: response.usage.candidateTokens,
        totalTokens: response.usage.totalTokens,
      } : undefined,
      processingTime,
    };
  } catch (error) {
